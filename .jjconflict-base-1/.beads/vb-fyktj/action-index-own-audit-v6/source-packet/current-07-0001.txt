36:     /// Appends one event to the LSM memtable without forcing a durability barrier.
37:     ///
38:     /// The event is committed to the WAL-backed LSM memtable and is visible
39:     /// to subsequent readers (replay / `events_for_run`) immediately on
40:     /// return.  "Unfsynced" is precise: the write survives process-level
41:     /// recovery (crash + restart reads back the memtable) but has not been
42:     /// force-flushed to stable storage.  Callers that require strict
43:     /// durability must invoke `persist_strict` after staging.
44:     ///
45:     /// The runtime journal path uses this entry point for direct appends.
46:     /// To keep recovery scans consistent with the durable event log,
47:     /// derived index mutations are staged into the SAME
48:     /// `fjall::OwnedWriteBatch` as the event write and committed
49:     /// atomically. A process crash between the event insert and an index
50:     /// mutation is impossible because they share one batch.
51:     pub(crate) fn append_unfsynced(&self, event: &JournalEvent) -> Result<(), JournalError> {
52:         let _guard = self
53:             .write_lock
54:             .lock()
55:             .map_err(|_| JournalError::WriteLockPoisoned)?;
56:         let key = run_event_key(event.run_id(), event.seq())?;
57:         if !event.is_valid() {
58:             return Err(JournalError::InvalidEvent);
59:         }
60:         if self.events.contains_key(key)? {
61:             return Err(JournalError::DuplicateEvent {
62:                 run: event.run_id(),
63:                 seq: event.seq(),
64:             });
65:         }
66:         let value = encode_record(
67:             MAGIC_JOURNAL_EVENT,
68:             event.record_kind(),
69:             event.seq().get(),
70:             event,
71:             MAX_JOURNAL_EVENT_PAYLOAD_BYTES,
72:         )?;
73:         // Stage event + recovery indexes on one batch so each derived
74:         // index mutation succeeds or fails with the event write.
75:         let mut batch = self.database.batch();
76:         self.stage_recovery_index_ops(&mut batch, event)?;
77:         batch.insert(&self.events, key, value);
78:         batch.commit()?;
79:         Ok(())
80:     }
