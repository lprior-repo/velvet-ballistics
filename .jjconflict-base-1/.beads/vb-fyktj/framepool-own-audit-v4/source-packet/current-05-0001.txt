72: fn bh_shd_04_find_error_handler_linear_scan_fallback() {
73:     let handler_idx = 20u16;
74:     let first_node = CompiledNode {
75:         id: vb_core::ids::StepIdx::ZERO,
76:         output: None,
77:         next: None,
78:         on_error: None,
79:         error_slot: None,
80:         kind: CompiledNodeKind::ErrorHandler {
81:             body: vb_core::ids::StepIdx::new(1),
82:             handler: vb_core::ids::StepIdx::new(handler_idx),
83:             error_slot: None,
84:         },
85:     };
86:     let middle_nodes = (1u16..handler_idx).map(|i| CompiledNode {
87:         id: vb_core::ids::StepIdx::new(i),
88:         output: None,
89:         next: Some(vb_core::ids::StepIdx::new(i + 1)),
90:         on_error: None,
91:         error_slot: None,
92:         kind: CompiledNodeKind::Nop,
93:     });
94:     let last_node = CompiledNode {
95:         id: vb_core::ids::StepIdx::new(handler_idx),
96:         output: Some(SlotIdx::new(0)),
97:         next: None,
98:         on_error: None,
99:         error_slot: None,
100:         kind: CompiledNodeKind::SetConst {
101:             value: ConstIdx::new(0),
102:         },
103:     };
104:     let nodes = std::iter::once(first_node)
105:         .chain(middle_nodes)
106:         .chain(std::iter::once(last_node))
107:         .collect::<Vec<_>>();
108:     let parts = WorkflowParts {
109:         semantic_source_identity: None,
110:         name: Box::from("bh_large_wf"),
111:         digest: WorkflowDigest::from_bytes([0xEE; 32]),
112:         nodes: nodes.into_boxed_slice(),
113:         expressions: Box::from([]),
114:         accessors: Box::from([]),
115:         constants: Box::from([vb_core::value::ConstValue::Bool(false)]),
116:         slot_count: 1,
117:         symbols_count: 0,
118:         entry: vb_core::ids::StepIdx::ZERO,
119:         step_names: Box::from([]),
120:         resource_contract: ResourceContract::DEFAULT,
121:         input_slots: Box::from([]),
122:     };
123:     let workflow = match vb_core::workflow::CompiledWorkflow::try_from_parts(parts) {
124:         Ok(w) => w,
125:         Err(_) => return,
126:     };
127:     let result = super::helpers::find_error_handler_for_failure(
128:         // The body step (step 1) is protected by the ErrorHandler at step 0.
129:         // Steps 2..N-1 are NOT protected since they are not the body.
130:         &workflow,
131:         vb_core::ids::StepIdx::new(1),
132:     );
133:     match result {
134:         Some((handler, _error_slot)) => {
135:             assert_eq!(
136:                 handler,
137:                 vb_core::ids::StepIdx::new(handler_idx),
138:                 "BH-SHD-04: linear scan should find handler at end of workflow"
139:             );
140:         }
141:         None => {
142:             panic!("BH-SHD-04: expected to find error handler via linear scan");
143:         }
144:     }
145: }
