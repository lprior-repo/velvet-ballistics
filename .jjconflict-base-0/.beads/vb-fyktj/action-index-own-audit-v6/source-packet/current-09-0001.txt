65: // PO-002: ActionScheduled writes event and index atomically (proptest)
66: // ---------------------------------------------------------------------------
67: 
68: proptest::proptest! {
69:     #![proptest_config(journal_proptest_config(JOURNAL_IO_PROPTEST_CASES))]
70: 
71:     /// PO-002: append_event + put_action_index staged together commit atomically.
72:     ///
73:     /// For each valid (action, run, step) combination:
74:     /// 1. Stage ActionScheduled event + action index entry in same batch
75:     /// 2. Commit batch
76:     /// 3. Verify journal events_for_run returns the ActionScheduled
77:     /// 4. Verify index_action keyspace contains entry for (action, run, step)
78:     #[test]
79:     fn test_action_scheduled_writes_event_and_index_atomically(
80:         action_val in 1u16..=1000u16,
81:         run_val in 1u64..=10000u64,
82:         step_val in 0u16..=100u16,
83:         seq_val in 0u64..=0u64,
84:     ) {
85:         let action = ActionId::new(action_val);
86:         let run = RunId::new(run_val);
87:         let step = StepIdx::new(step_val);
88:         let (_temp, journal) = temp_journal();
89: 
90:         let event = make_action_scheduled(run, seq_val, step, action);
91: 
92:         // Stage event + index entry in same atomic batch
93:         let mut batch = JournalWriteBatch::new(&journal);
94:         let append_result = batch.append_event(&event);
95:         prop_assert!(append_result.is_ok(), "append_event must succeed for valid event");
96: 
97:         let index_result = batch.put_action_index(action, run, step);
98:         prop_assert!(index_result.is_ok(), "put_action_index must succeed for valid inputs");
99: 
100:         let commit_result = batch.commit();
101:         prop_assert!(commit_result.is_ok(), "batch commit must succeed");
102: 
103:         // Verify event is durable in journal
104:         let events = journal.events_for_run(run)
105:             .expect("events_for_run must succeed");
106:         prop_assert!(events.len() == 1, "exactly 1 event must be durable after batch commit");
107:         prop_assert_eq!(&events[0], &event, "durable event must match committed event");
108: 
109:         // Verify index entry is durable — scan index_action keyspace
110:         let index_key = vb_storage::keys::index_action_key(action, run, step)
111:             .expect("index_action_key must succeed for valid inputs");
112:         let has_entry = journal.has_action_index_entry(index_key)
113:             .expect("has_action_index_entry must succeed");
114:         prop_assert!(has_entry, "index_action entry must be durable for (action, run, step)");
115:     }
116: }
