49: /// Lossless conversion of a `u32` integer to its exact `f32` representation.
50: ///
51: /// `From<u32> for f32` is NOT implemented by the Rust standard library
52: /// (only `From<u8>`, `From<u16>`, `From<i8>`, `From<i16>` exist for `f32`).
53: /// For values in `[0, 2^24)` — which includes the full RA-003 trace-ring
54: /// domain (`cap <= 2^20`, `len <= cap`) — the IEEE-754 single-precision
55: /// encoding fits the integer exactly, so this helper produces a result
56: /// bit-identical to `(n as f32)` without using an `as`-cast and without
57: /// tripping `clippy::as_conversions`. All integer arithmetic uses
58: /// `u32::checked_*` / `u32::saturating_*` so `clippy::arithmetic_side_effects`
59: /// is also satisfied.
60: fn u32_to_f32_exact(n: u32) -> f32 {
61:     if n == 0 {
62:         return 0.0_f32;
63:     }
64:     // `e = floor(log2(n))`. For n in [1, 2^32), `leading_zeros` is in [0, 31],
65:     // so `e` is in [0, 31]. The `31 - ...` formula is the bit-width (32) minus
66:     // 1 (for the implicit leading one) minus `leading_zeros`.
67:     let e = u32::checked_sub(31, n.leading_zeros()).unwrap_or(0);
68:     let biased_exp = u32::saturating_add(e, 127);
69:     let power = 1_u32.checked_shl(e).unwrap_or(1);
70:     let mantissa = u32::checked_sub(n, power)
71:         .unwrap_or(0)
72:         .checked_shl(23_u32.saturating_sub(e))
73:         .unwrap_or(0);
74:     f32::from_bits((biased_exp << 23) | mantissa)
75: }

198:     /// Creates a new runtime with the given number of shards, per-shard configuration, and
199:     /// explicit [`SharedRuntimeJournal`].
200:     ///
201:     /// The journal argument is **required**: this constructor will not silently pick a
202:     /// non-durable default. Master §18 requires Fjall-backed persistence for recovery,
203:     /// so production callers must wire a [`crate::journal::StorageRuntimeJournal`] (strict
204:     /// barrier) or [`crate::journal::QueuedStorageRuntimeJournal`] (group-committed).
205:     /// Tests and benchmarks that do not need durability should pass
206:     /// [`crate::journal::VolatileRuntimeJournal::shared`] or
207:     /// [`crate::journal::NoopRuntimeJournal::shared_for_tests_and_benchmarks`] so the
208:     /// call site documents the non-durable intent.
209:     #[must_use]
210:     pub fn new(
211:         shard_count: NonZeroUsize,
212:         config: ShardConfig,
213:         journal: SharedRuntimeJournal,
214:     ) -> Self {
215:         let count = shard_count.get();
216:         let mut shards = Vec::with_capacity(count);
217:         let mut index = 0usize;
218:         while index < count {
219:             shards.push(Shard::new_with_journal(config, journal.clone()));
220:             index = index.saturating_add(1);
221:         }
222:         Self {
223:             shards,
224:             shard_count: count,
225:             journal,
226:             policy: config.policy,
227:             live_policy: crate::policy::LivePolicy::new(),
228:         }
229:     }

399:     /// Recovers a run from durable Fjall journal evidence and enqueues it for resumption.
400:     ///
401:     /// This is the end-to-end recovery entry point: it reads the journal,
402:     /// replays events to reconstruct a typed `RecoveryFrameSeedProduct`,
403:     /// hydrates the recoverable live-frame portion, restores durable
404:     /// pending-action authority when present, and enqueues
405:     /// `ShardCommand::Recover` on the owning shard.
406:     ///
407:     /// Recovery crosses the bead `vb-w25-runtime-a2` FINDING-001/FINDING-002
408:     /// typestate boundary: the storage layer never hands back a raw
409:     /// `RecoveryFrameSeed`. It returns the typed
410:     /// `vb_storage::recovery::RecoveryFrameSeedProduct` (the storage
411:     /// cannot-resume/resumable witness, FINDING-002), which
412:     /// `recover_product` classifies through the parallel runtime boundary
413:     /// into a `crate::recovery::RuntimeRecoveryProduct`
414:     /// (`SummaryOnly` / `CannotResume { reason }` / `Resumable`,
415:     /// FINDING-001). `recover_and_resume` enqueues only the `Resumable`
416:     /// product and rejects the rest with a typed error.
417:     ///
418:     /// Returns `RuntimeError::RecoveryNotAvailable` when the runtime's
419:     /// journal is not storage-backed (noop or volatile journals cannot
420:     /// be replayed for crash recovery).
421:     ///
422:     /// Returns `RuntimeError::Recovery` when the recovery operation fails
423:     /// the journal replay fails (missing data, corrupt snapshot, digest
424:     /// mismatch, or unsupported recovery state).
425:     pub fn recover_and_resume(&self, run: RunId) -> RuntimeResult<()> {
426:         match self.recover_product(run)? {
427:             crate::recovery::RuntimeRecoveryProduct::Resumable(product) => {
428:                 self.enqueue_recovered_product(*product)
429:             }
430:             other => Err(Self::rejected_recovery_product_error(other)),
431:         }
432:     }

434:     /// Recovers a failed run from a validated retry checkpoint.
435:     ///
436:     /// Unlike [`Runtime::recover_and_resume`], this path uses the full
437:     /// [`crate::run_checkpoint::RunCheckpointV1`] captured at the failure boundary,
438:     /// rather than a legacy resumable frame seed. The checkpoint identity is checked
439:     /// against the run before the recovery command is enqueued; `handle_recover`
440:     /// re-validates the checkpoint against the independently loaded accepted artifact
441:     /// (artifact digest, workflow digest, policy, ABI grant authority, frame/store/collect
442:     /// integrity) before any runtime state mutation.
443:     pub fn recover_retry(
444:         &self,
445:         run: RunId,
446:         checkpoint: crate::run_checkpoint::RunCheckpointV1,
447:         next_seq: vb_storage::EventSeq,
448:     ) -> RuntimeResult<()> {
449:         if checkpoint.run_id != run {
450:             return Err(RuntimeError::Recovery {
451:                 error: "retry checkpoint run identity mismatch".to_string(),
452:             });
453:         }
454:         // Bind the caller-supplied checkpoint and next_seq to the actual durable chain
455:         // (latest accepted RetryCheckpoint + its preceding RunFailed event). A legacy,
456:         // missing, tampered, stale, or cross-run checkpoint fails closed here, before any
457:         // state or journal mutation. The CLI boundary is digest-protected, but this public
458:         // raw API must not trust caller-supplied run-attempt / prior-seq / next-seq authority.
459:         self.validate_retry_checkpoint_against_durable_chain(run, &checkpoint, next_seq)?;
460:         let command = crate::shard::types::RecoverRunCommand::for_retry(checkpoint, next_seq)?;
461:         let shard = self.shard_for(run)?;
462:         shard.enqueue(ShardCommand::Recover {
463:             command: Box::new(command),
464:         })
465:     }
466: 
467:     /// Validates a caller-supplied retry checkpoint against the run's durable journal.
468:     ///
469:     /// The durable failure boundary ends in `RunFailedEvent { seq: S, attempt: A }` followed by
470:     /// `RetryCheckpoint { seq: S+1, attempt: A+1 }`. The supplied checkpoint must decode to the
471:     /// exact latest durable `RetryCheckpoint` payload (binding run/step/attempt/prior_seq/
472:     /// artifact/workflow/policy/grants/frame/store/collect/contracts), its `prior_seq` must name
473:     /// the immediately preceding `RunFailedEvent` sequence, and `next_seq` must be the sequence
474:     /// immediately after the checkpoint. Checked arithmetic throughout; every rejection is
475:     /// fail-closed with no state or journal mutation.
476:     fn validate_retry_checkpoint_against_durable_chain(
477:         &self,
478:         run: RunId,
479:         checkpoint: &crate::run_checkpoint::RunCheckpointV1,
480:         next_seq: vb_storage::EventSeq,
481:     ) -> RuntimeResult<()> {
482:         let journal = self.recovery_storage_journal()?;
483:         let events = journal
484:             .events_for_run_full(run)
485:             .map_err(|error| RuntimeError::Recovery {
486:                 error: format!("retry journal read failed: {error}"),
487:             })?;
488: 
489:         let durable = events.iter().rev().find(|event| {
490:             matches!(
491:                 event,
492:                 vb_storage::JournalEvent::RetryCheckpoint { run: r, .. } if *r == run
493:             )
494:         });
495:         let vb_storage::JournalEvent::RetryCheckpoint {
496:             run: durable_run,
497:             seq: durable_seq,
498:             attempt: durable_attempt,
499:             payload_digest: durable_payload_digest,
500:             payload: durable_payload,
501:             ..
502:         } = durable.ok_or_else(|| RuntimeError::Recovery {
503:             error: "run has no durable retry checkpoint".to_string(),
504:         })?
505:         else {
506:             return Err(RuntimeError::Recovery {
507:                 error: "run has no durable retry checkpoint".to_string(),
508:             });
509:         };
510: 
511:         // Integrity of the durable record itself.
512:         let observed_digest = *blake3::hash(durable_payload).as_bytes();
513:         if observed_digest != *durable_payload_digest {
514:             return Err(RuntimeError::Recovery {
515:                 error: "durable retry checkpoint payload digest mismatch".to_string(),
516:             });
517:         }
518: 
519:         // The supplied checkpoint must decode to the exact durable checkpoint value.
520:         // Use the version-aware decoder to reject historical version 1 payloads
521:         // that lack the policy_pin field in RunAdmission.
522:         let durable_checkpoint = crate::run_checkpoint::decode_checkpoint_payload(durable_payload)
523:             .map_err(|error| RuntimeError::Recovery {
524:                 error: format!("durable retry checkpoint decode failed: {error}"),
525:             })?;
526:         if durable_checkpoint != *checkpoint {
527:             return Err(RuntimeError::Recovery {
528:                 error: "retry checkpoint does not match durable RetryCheckpoint evidence"
529:                     .to_string(),
530:             });
531:         }
532: 
533:         // run_attempt / step authority (also covered by the value equality above, but these
534:         // yield clearer diagnostics and double as a defense-in-depth binding).
535:         if *durable_run != run {
536:             return Err(RuntimeError::Recovery {
537:                 error: "durable retry checkpoint belongs to a different run".to_string(),
538:             });
539:         }
540:         if checkpoint.run_attempt.get() != *durable_attempt {
541:             return Err(RuntimeError::Recovery {
542:                 error: "retry checkpoint run attempt does not match durable RetryCheckpoint"
543:                     .to_string(),
544:             });
545:         }
546: 
547:         // prior_seq must be the sequence immediately before the durable checkpoint.
548:         let expected_checkpoint_seq =
549:             checkpoint
550:                 .prior_seq
551:                 .checked_add(1)
552:                 .ok_or_else(|| RuntimeError::Recovery {
553:                     error: "retry checkpoint prior_seq overflow".to_string(),
554:                 })?;
555:         if durable_seq.get() != expected_checkpoint_seq {
556:             return Err(RuntimeError::Recovery {
557:                 error: "retry checkpoint prior_seq does not match durable RetryCheckpoint position"
558:                     .to_string(),
559:             });
560:         }
561: 
562:         // next_seq must be the sequence immediately after the durable checkpoint.
563:         let expected_next_seq =
564:             durable_seq
565:                 .get()
566:                 .checked_add(1)
567:                 .ok_or_else(|| RuntimeError::Recovery {
568:                     error: "retry checkpoint next_seq overflow".to_string(),
569:                 })?;
570:         if next_seq.get() != expected_next_seq {
571:             return Err(RuntimeError::Recovery {
572:                 error: "retry next_seq does not match durable RetryCheckpoint position".to_string(),
573:             });
574:         }
575: 
576:         // The checkpoint must be the journal head. An explicit retry that already consumed
577:         // this checkpoint appends StepStarted / ActionScheduled / RunFinished AFTER it, so a
578:         // checkpoint that is no longer the last event is already-consumed and must be rejected
579:         // (a stale `next_seq` alone cannot be trusted to detect this).
580:         let checkpoint_is_head = events
581:             .last()
582:             .is_some_and(|event| event.seq().get() == durable_seq.get());
583:         if !checkpoint_is_head {
584:             return Err(RuntimeError::Recovery {
585:                 error: "retry checkpoint has already been consumed (journal advanced past it)"
586:                     .to_string(),
587:             });
588:         }
589: 
590:         // The event immediately preceding the checkpoint must be the run's own RunFailed
591:         // event for the attempt immediately before the checkpoint's run_attempt.
592:         let expected_failed_attempt =
593:             checkpoint
594:                 .run_attempt
595:                 .get()
596:                 .checked_sub(1)
597:                 .ok_or_else(|| RuntimeError::Recovery {
598:                     error: "retry checkpoint run_attempt underflow".to_string(),
599:                 })?;
600:         let preceding_failed = events.iter().find(|event| {
601:             event.seq().get() == checkpoint.prior_seq
602:                 && matches!(
603:                     event,
604:                     vb_storage::JournalEvent::RunFailedEvent { run: r, .. } if *r == run
605:                 )
606:         });
607:         match preceding_failed {
608:             Some(vb_storage::JournalEvent::RunFailedEvent { attempt, .. })
609:                 if *attempt == expected_failed_attempt =>
610:             {
611:                 Ok(())
612:             }
613:             _ => Err(RuntimeError::Recovery {
614:                 error: "retry checkpoint does not follow the run's own durable RunFailed event"
615:                     .to_string(),
616:             }),
617:         }
618:     }

882:     fn product_from_recovered_context(
883:         context: RecoveredSeedProductContext,
884:         retained_inputs: Option<vb_storage::RunInputsRecord>,
885:     ) -> RuntimeResult<crate::recovery::RuntimeRecoveryProduct> {

1020:     /// Processes one command on each shard. Returns false if any shard is shutting down.
1021:     pub fn tick_all(&mut self) -> RuntimeResult<bool> {
1022:         let mut alive = true;
1023:         for shard in &mut self.shards {

1025:                 alive = false;
1026:             }
1027:         }

1031:     /// Processes one tick on a specific shard with the given directive.
1032:     ///
1033:     /// This method directs a single shard's behavior for one tick. The directive
1034:     /// determines what work the shard performs:
1035:     ///
1036:     /// - `Continue`: Process one command from the queue normally.
1037:     /// - `Suspend`: Skip command processing; preserve queue depth.
1038:     /// - `Migrate { target }`: Transfer all pending commands to the target shard.
1039:     /// - `Shutdown`: Drain all remaining commands and shut down the shard.
1040:     ///
1041:     /// Returns `Ok(true)` if the shard is alive (continuing), `Ok(false)` if the
1042:     /// shard has shut down, or an error if the shard index is invalid or migration
1043:     /// failed.
1044:     pub fn tick_shard(
1045:         &mut self,
1046:         shard_index: u32,
1047:         directive: ShardDirective,
1048:     ) -> RuntimeResult<bool> {
1049:         let shard_index_usize = usize::try_from(shard_index)
1050:             .map_err(|_| RuntimeError::ShardNotFound { shard: shard_index })?;
1051: 
1052:         // Validate source shard exists first
1053:         if self.shards.get(shard_index_usize).is_none() {
1054:             return Err(RuntimeError::ShardNotFound { shard: shard_index });
1055:         }
1056: 
1057:         match directive {
1058:             ShardDirective::Continue => {
1059:                 let shard = self
1060:                     .shards
1061:                     .get_mut(shard_index_usize)
1062:                     .ok_or(RuntimeError::ShardNotFound { shard: shard_index })?;
1063:                 shard.tick()
1064:             }
1065:             ShardDirective::Suspend => {
1066:                 // Suspend: skip processing, preserve queue, return alive
1067:                 Ok(true)
1068:             }
1069:             ShardDirective::Migrate { target } => self.migrate_shard(shard_index_usize, target),
1070:             ShardDirective::Shutdown => {
1071:                 let shard = self
1072:                     .shards
1073:                     .get_mut(shard_index_usize)
1074:                     .ok_or(RuntimeError::ShardNotFound { shard: shard_index })?;
1075:                 shard.drain_pending_and_shutdown()?;
1076:                 Ok(false)
1077:             }
1078:             ShardDirective::Cancel => Err(RuntimeError::UnsupportedOperation {
1079:                 operation: "tick_shard_cancel",
1080:             }),
1081:             ShardDirective::Barrier => Err(RuntimeError::UnsupportedOperation {
1082:                 operation: "tick_shard_barrier",
1083:             }),
1084:         }
1085:     }
1086: 
1087:     /// Migrates all commands from source shard to target shard.
1088:     ///
1089:     /// Returns `Ok(true)` if the source shard is still alive (has runs or pending commands),
1090:     /// `Ok(false)` if the source shard is empty and can be shut down.
1091:     fn migrate_shard(&mut self, source_idx: usize, target: u32) -> RuntimeResult<bool> {
1092:         let target_usize =
1093:             usize::try_from(target).map_err(|_| RuntimeError::ShardNotFound { shard: target })?;
1094: 
1095:         // Self-migrate check (already validated in caller, but double-check for safety)
1096:         let source_u32 =
1097:             u32::try_from(source_idx).map_err(|_| RuntimeError::ShardNotFound { shard: target })?;
1098:         if target == source_u32 {
1099:             return Err(RuntimeError::MigrateSelf);
1100:         }
1101: 
1102:         // Validate target shard exists (already validated in caller, but double-check)
1103:         if self.shards.get(target_usize).is_none() {
1104:             return Err(RuntimeError::ShardNotFound { shard: target });
1105:         }
1106: 
1107:         // Collect all commands from source shard
1108:         let commands: Vec<ShardCommand> = {
1109:             let shard = self
1110:                 .shards
1111:                 .get_mut(source_idx)
1112:                 .ok_or(RuntimeError::ShardNotFound { shard: source_u32 })?;
1113:             let mut cmds = Vec::new();
1114:             while let Some(cmd) = shard.command_queue.pop() {
1115:                 cmds.push(cmd);
1116:             }
1117:             cmds
1118:         };
1119: 
1120:         // Push all commands to target shard
1121:         {
1122:             let target_shard = self
1123:                 .shards
1124:                 .get_mut(target_usize)
1125:                 .ok_or(RuntimeError::ShardNotFound { shard: target })?;
1126:             for cmd in commands {
1127:                 target_shard.enqueue(cmd)?;
1128:             }
1129:         }
1130: 
1131:         // Return true if source shard still has runs (alive)
1132:         let shard = self
1133:             .shards
1134:             .get_mut(source_idx)
1135:             .ok_or(RuntimeError::ShardNotFound { shard: source_u32 })?;
1136:         let alive = shard.active_run_count() > 0 || !shard.command_queue.is_empty();
1137:         Ok(alive)
1138:     }

1175:     /// Looks up the canonical in-flight action ticket for `run`.
1176:     ///
1177:     /// IPC handlers receive only a `(run_id, ticket: u64)` wire pair. The wire
1178:     /// `ticket` encodes only the lower-16-bit step index and does not carry the
1179:     /// real `seq`, `action`, `attempt`, `idempotency_key`, or `capacity`
1180:     /// values. The canonical ticket is held by the shard's `pending_actions`
1181:     /// index and is the only authoritative source of those fields.
1182:     ///
1183:     /// Returns `None` if:
1184:     /// - the run is not found,
1185:     /// - the run has no in-flight action ticket,
1186:     /// - the wire `ticket`'s lower 16 bits do not match the canonical step.
1187:     ///
1188:     /// This lookup is the boundary that prevents the IPC layer from
1189:     /// fabricating dummy `seq`/`action`/`idempotency_key`/`capacity` fields
1190:     /// (see bead vb-xb62s). Callers must fail closed if this returns `None`
1191:     /// and must never invent fields the wire did not carry.
1192:     #[must_use]
1193:     pub fn lookup_pending_action_ticket(
1194:         &self,
1195:         run: RunId,
1196:         ticket_wire: u64,
1197:     ) -> Option<vb_core::action::ActionTicket> {
1198:         let shard = self.shard_for_run(run).ok()?;
1199:         let canonical = shard.pending_action_get(run)?;
1200:         let wire_step = u16::try_from(ticket_wire).ok()?;
1201:         if canonical.step.get() != wire_step {
1202:             return None;
1203:         }
1204:         Some(canonical)
1205:     }

1296:     /// Collects runtime metrics from all shards.
1297:     pub fn collect_metrics(&self) -> RuntimeMetricsSnapshot {
1298:         let mut shards = Vec::with_capacity(self.shard_count);
1299:         let mut runs_active = 0u32;
1300:         let mut runs_waiting = 0u32;
1301:         let mut runs_failed_total = 0u64;
1302:         let mut runs_finished_total = 0u64;
1303:         let mut steps_total = 0u64;
1304: 
1305:         for (index, shard) in self.shards.iter().enumerate() {
1306:             let counters = shard.counters().snapshot();
1307:             let active_runs = u32::try_from(shard.active_run_count()).unwrap_or(u32::MAX);
1308:             let queue_depth = u32::try_from(shard.command_queue_len()).unwrap_or(u32::MAX);
1309:             let queue_remaining = u32::try_from(shard.remaining_capacity()).unwrap_or(u32::MAX);
1310:             let pending_timers = u32::try_from(shard.pending_timer_count()).unwrap_or(u32::MAX);
1311:             let (fp_free, fp_total) = shard.frame_pool_metrics();
1312:             let frame_pool_free = u32::try_from(fp_free).unwrap_or(u32::MAX);
1313:             let frame_pool_total = u32::try_from(fp_total).unwrap_or(u32::MAX);
1314:             let trace_capacity = shard.trace_ring().capacity();
1315:             let trace_len = shard.trace_ring().pending_len();
1316:             let trace_ring_fill_pct = if trace_capacity > 0 {
1317:                 // Bounded narrowing mirrors the six sibling metric lines at runtime.rs:571-577.
1318:                 // TraceRing::new clamps capacity to >= 1 (RA-003 cap bound 2^20 << 2^24), so the
1319:                 // unwrap_or(0) fallback is unreachable in production. Fallback value is 0 (not
1320:                 // u32::MAX) to preserve the sentinel intent of the outer zero-denominator guard.
1321:                 //
1322:                 // DEVIATION FROM CONTRACT INV-004: `f32::from(u32)` is NOT implemented in
1323:                 // Rust (only `From<u8|u16|i8|i16>` exist for f32). See `u32_to_f32_exact`
1324:                 // above for the bit-equivalent IEEE-754 manual encoding; equivalence to
1325:                 // `(n as f32)` is verified in `.beads/vb-oul6u/evidence/ieee-754-bit-equivalence.log`
1326:                 // and pinned by the RA-003 corpus (`trace_ring_fill_pct` tests, 3/3 pass).
1327:                 let cap_u32 = u32::try_from(trace_capacity).unwrap_or(0);
1328:                 let len_u32 = u32::try_from(trace_len).unwrap_or(0);
1329:                 let len_f32 = u32_to_f32_exact(len_u32);
1330:                 let cap_f32 = u32_to_f32_exact(cap_u32);
1331:                 (len_f32 / cap_f32) * 100.0
1332:             } else {
1333:                 0.0
1334:             };
1335: 
1336:             runs_active = runs_active.saturating_add(active_runs);
1337:             runs_waiting = runs_waiting.saturating_add(pending_timers);
1338:             runs_failed_total = runs_failed_total.saturating_add(counters.runs_failed);
1339:             runs_finished_total = runs_finished_total.saturating_add(counters.runs_completed);
1340:             steps_total = steps_total.saturating_add(counters.steps_executed);
1341: 
1342:             let shard_id = u32::try_from(index).unwrap_or(u32::MAX);
1343:             shards.push(ShardMetricsSnapshot {
1344:                 shard_id,
1345:                 active_runs,
1346:                 command_queue_depth: queue_depth,
1347:                 command_queue_remaining: queue_remaining,
1348:                 pending_timers,
1349:                 frame_pool_free,
1350:                 frame_pool_total,
1351:                 trace_ring_fill_pct,
1352:                 counters,
1353:             });
1354:         }
1355: 
1356:         RuntimeMetricsSnapshot {
1357:             shards,
1358:             runs_active,
1359:             runs_waiting,
1360:             runs_failed_total,
1361:             runs_finished_total,
1362:             steps_total,
1363:         }

1384:     /// Lists active run summaries across all shards, up to `limit` entries.
1385:     ///
1386:     /// If `workflow_filter` is provided, only runs matching that digest are included.
1387:     /// Returns summaries sorted by run id ascending.
1388:     pub fn list_active_runs(
1389:         &self,
1390:         limit: u32,
1391:         workflow_filter: Option<vb_core::WorkflowDigest>,
1392:     ) -> Vec<ActiveRunSummary> {
1393:         let max = match usize::try_from(limit) {
1394:             Ok(value) => value,
1395:             Err(_) => usize::MAX,
1396:         };
1397:         let mut summaries = Vec::new();
1398:         for shard in &self.shards {
1399:             for (run_id, state) in shard.active_runs_iter() {
1400:                 if summaries.len() >= max {
1401:                     break;
1402:                 }
1403:                 let digest = state.workflow.digest();
1404:                 if let Some(filter) = workflow_filter
1405:                     && digest != filter
1406:                 {
1407:                     continue;
1408:                 }
1409:                 let step_count = state.workflow.node_count();
1410:                 let mut steps_completed: u16 = 0;
1411:                 let mut step_index = 0u16;
1412:                 while step_index < step_count {
1413:                     let step = vb_core::ids::StepIdx::new(step_index);
1414:                     match state.frame.step_state(step) {
1415:                         Ok(
1416:                             vb_core::frame::StepState::Succeeded
1417:                             | vb_core::frame::StepState::Failed
1418:                             | vb_core::frame::StepState::Skipped
1419:                             | vb_core::frame::StepState::Cancelled,
1420:                         ) => {
1421:                             steps_completed = steps_completed.saturating_add(1);
1422:                         }
1423:                         Ok(_) => {}
1424:                         Err(_) => {
1425:                             step_index = step_index.saturating_add(1);
1426:                             continue;
1427:                         }
1428:                     }
1429:                     step_index = step_index.saturating_add(1);
1430:                 }
1431:                 summaries.push(ActiveRunSummary {
1432:                     run_id: *run_id,
1433:                     workflow: digest,
1434:                     step_count,
1435:                     steps_completed,
1436:                 });
1437:             }
1438:         }
1439:         summaries.sort_by_key(|s| s.run_id);
1440:         summaries.truncate(max);
1441:         summaries
1442:     }

1477:     /// Returns the shard that currently owns `run`, scanning all shards.
1478:     ///
1479:     /// RA-030 wave-15 (vb-sxkz6): the hash-based `shard_index(run)` returns
1480:     /// the home shard which may differ from the owner after `migrate_shard`
1481:     /// transfers a run's commands and run-state to a different shard. We
1482:     /// check the home shard first as a fast path; on miss we scan the
1483:     /// remaining shards in deterministic slice order.
1484:     ///
1485:     /// Returns `Err(RuntimeError::RunNotFound)` if `run` is not in any
1486:     /// shard's `run_state` or `terminal_runs` set.
1487:     fn shard_for_run(&self, run: RunId) -> Result<&Shard, RuntimeError> {
1488:         let home_index = self.shard_index(run);
1489:         if let Some(shard) = self.shards.get(home_index)
1490:             && (shard.run_state_contains(run) || shard.terminal_runs_contains(run))
1491:         {
1492:             return Ok(shard);
1493:         }
1494:         self.shards
1495:             .iter()
1496:             .enumerate()
1497:             .find(|(idx, shard)| {
1498:                 *idx != home_index
1499:                     && (shard.run_state_contains(run) || shard.terminal_runs_contains(run))
1500:             })
1501:             .map(|(_, shard)| shard)
1502:             .ok_or(RuntimeError::RunNotFound)
1503:     }

1505:     /// Mutable variant of `shard_for_run` for `take_inspect_response`.
1506:     fn shard_for_run_mut(&mut self, run: RunId) -> Result<&mut Shard, RuntimeError> {
1507:         // Mirror the home-fast-path + scan-fallback logic on immutable borrow,
1508:         // then re-index after borrow ends to get the mutable shard.
1509:         let home_index = self.shard_index(run);
1510:         let owner_index = {
1511:             if let Some(shard) = self.shards.get(home_index)
1512:                 && (shard.run_state_contains(run) || shard.terminal_runs_contains(run))
1513:             {
1514:                 home_index
1515:             } else {
1516:                 self.shards
1517:                     .iter()
1518:                     .position(|shard| {
1519:                         shard.run_state_contains(run) || shard.terminal_runs_contains(run)
1520:                     })
1521:                     .ok_or(RuntimeError::RunNotFound)?
1522:             }
1523:         };
1524:         self.shards
1525:             .get_mut(owner_index)
1526:             .ok_or(RuntimeError::RunNotFound)
1527:     }
