#![forbid(unsafe_code)]

use std::fs::File;
use std::io::{BufReader, Write};
use std::path::PathBuf;

use vb_core::admission::AcceptedArtifact;
use vb_core::policy::RuntimePolicy;
use vb_core::value::ConstValue;
use vb_core::workflow::{CompiledWorkflow, WorkflowParts};
use vb_storage::admission::{
    AcceptedArtifact, VerificationWarning, accepted_artifact_digest, submit_artifact,
};
use vb_storage::journal::FjallJournal;

fn temp_journal() -> Result<FjallJournal, String> {
    let dir = PathBuf::from(format!(
        "/tmp/vb-storage-test-{}",
        std::time::SystemTime::now()
            .duration_since(std::time::UNIX_EPOCH)
            .unwrap()
            .as_nanos()
    ));
    std::fs::create_dir_all(&dir).map_err(|e| format!("create temp dir: {e}"))?;
    FjallJournal::open(dir).map_err(|e| format!("open journal: {e}"))
}

fn minimal_workflow() -> Result<CompiledWorkflow, String> {
    let parts = WorkflowParts {
        name: "test".to_string(),
        steps: vec![],
    };
    CompiledWorkflow::compile(&parts).map_err(|e| format!("compile: {e}"))
}

fn submit_minimal(policy: RuntimePolicy) -> Result<AcceptedArtifact, String> {
    let journal = temp_journal()?;
    let workflow = minimal_workflow()?;
    submit_artifact(&journal, &workflow, policy)
        .map_err(|e| format!("submit failed: {e}"))
}

fn warning_at(gate: u8) -> VerificationWarning {
    VerificationWarning::new(gate, "test warning".to_string())
}

#[test]
fn accepted_artifact_validator_accepts_warning_gate_fifteen() -> Result<(), String> {
    let artifact = submit_minimal(RuntimePolicy::Strict)?;
    let warnings = vec![warning_at(15)];
    AcceptedArtifact::validate(&artifact, &warnings)?;
    Ok(())
}

#[test]
fn accepted_artifact_validator_rejects_warning_gate_sixteen() -> Result<(), String> {
    let artifact = submit_minimal(RuntimePolicy::Strict)?;
    let warnings = vec![warning_at(16)];
    let result = AcceptedArtifact::validate(&artifact, &warnings);
    assert!(result.is_err());
    Ok(())
}

#[test]
fn accepted_artifact_validator_uses_fifteen_gate_v1_upper_bound() {
    assert_eq!(VerificationWarning::MAX_GATE, 15);
}

#[test]
fn accepted_artifact_validator_rejects_legacy_thirteen_gate_upper_bound() {
    assert_ne!(VerificationWarning::MAX_GATE, 13);
}

#[test]
fn accepted_artifact_store_payload_is_raw_workflow_parts_not_nested_artifact() -> Result<(), String>
{
    let journal = temp_journal()?;
    let workflow = minimal_workflow()?;
    let artifact = submit_artifact(&journal, &workflow, RuntimePolicy::Strict)?;
    let stored = journal
        .compiled_ir(artifact.digest)
        .map_err(|e| format!("load failed: {e}"))?
        .ok_or_else(|| String::from("artifact not found"))?;
    // Stored payload is the compiled workflow bytes, not a nested artifact structure
    let payload = postcard::from_bytes::<CompiledWorkflow>(&stored).map_err(|e| format!("decode stored payload: {e}"))?;
    assert_eq!(payload.name, "test");
    Ok(())
}

#[test]

        .map_err(|e| format!("load failed: {e}"))?
        .ok_or_else(|| String::from("artifact not found"))?;

    Ok(())
}

#[test]
fn accepted_artifact_roundtrip_through_storage_persists_and_loads() -> Result<(), String> {
    let journal = temp_journal()?;
    let workflow = minimal_workflow()?;
    let artifact = submit_artifact(&journal, &workflow, RuntimePolicy::Strict)
        .map_err(|e| format!("submit failed: {e}"))?;

    let stored = journal
        .compiled_ir(artifact.digest)
        .map_err(|e| format!("load failed: {e}"))?
        .ok_or_else(|| String::from("artifact not found after submit"))?;

    let loaded: CompiledWorkflow = postcard::from_bytes(&stored)
        .map_err(|e| format!("decode failed: {e}"))?;
    assert_eq!(loaded.name, "test");
    Ok(())
}

#[test]
fn accepted_artifact_stored_bytes_are_postcard_encoded() -> Result<(), String> {
    let journal = temp_journal()?;
    let workflow = minimal_workflow()?;
    let artifact = submit_artifact(&journal, &workflow, RuntimePolicy::Strict)
        .map_err(|e| format!("submit failed: {e}"))?;

    let stored = journal
        .compiled_ir(artifact.digest)
        .map_err(|e| format!("load failed: {e}"))?
        .ok_or_else(|| String::from("artifact not found"))?;

    // Verify postcard encoding by roundtripping
    let decoded: CompiledWorkflow = postcard::from_bytes(&stored)
        .map_err(|e| format!("postcard decode failed: {e}"))?;
    assert_eq!(decoded.name, "test");
    Ok(())
}
