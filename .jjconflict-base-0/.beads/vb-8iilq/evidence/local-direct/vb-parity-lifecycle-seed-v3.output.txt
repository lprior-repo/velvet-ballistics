#!/usr/bin/env python3

def find_function_range(source, func_name):
    pattern = rf'fn\s+{re.escape(func_name)}\s*\('
    match = re.search(pattern, source)
    if not match:
        return None
    start = match.start()
    i = match.end()
    paren_depth = 1
    while i < len(source):
        if source[i] == '(':
            paren_depth += 1
        elif source[i] == ')':
            paren_depth -= 1
            if paren_depth == 0:
                break
        i += 1
    while i < len(source) and source[i] != '{':
        i += 1
    if i >= len(source):
        return None
    brace_depth = 0
    while i < len(source):
        if source[i] == '{':
            brace_depth += 1
        elif source[i] == '}':
            brace_depth -= 1
            if brace_depth == 0:
                return (start, i + 1)
        i += 1
    return None

def find_submit_exprs(body):
    results = []
    variants = [
        'ShardCommand::SubmitWithInputsAndContracts',
        'ShardCommand::SubmitWithInputs',
        'ShardCommand::SubmitWithContracts',
        'ShardCommand::Submit',
    ]
    for variant in variants:
        idx = 0
        while True:
            idx = body.find(variant, idx)
            if idx == -1:
                break
            before_ok = idx == 0 or not (body[idx-1].isalnum() or body[idx-1] == '_')
            after = idx + len(variant)
            after_ok = after >= len(body) or body[after] in '{ '
            if before_ok and after_ok:
                brace_idx = body.find('{', idx)
                if brace_idx != -1:
                    brace_depth = 0
                    j = brace_idx
                    while j < len(body):
                        if body[j] == '{':
                            brace_depth += 1
                        elif body[j] == '}':
                            brace_depth -= 1
                            if brace_depth == 0:
                                results.append((idx, variant, brace_idx, j))
                                break
                        j += 1
            idx += 1
    return results

def transform_expr(body, expr_start, variant, brace_start, brace_end):
    old_expr = body[expr_start:brace_end+1]
    inner = body[brace_start+1:brace_end]
    has_caps = bool(re.search(r'\bcaps\s*:', inner))
    has_inputs = bool(re.search(r'\binputs\s*:', inner))
    has_contracts = bool(re.search(r'\baction_contracts\s*:', inner))
    new_inner = inner
    if has_caps:
        new_inner = re.sub(
            r'caps\s*:\s*CapabilitySet::empty\(\)',
            'caps: crate::shard::tests::test_fixtures::contract_capability_for(vb_core::ActionId::new(0))',
            new_inner
        )
    injections = []
    if not has_caps:
        injections.append('caps: crate::shard::tests::test_fixtures::contract_capability_for(vb_core::ActionId::new(0))')
    if not has_inputs:
        injections.append('inputs: Box::from([(SlotIdx::ZERO, SlotValue::I64(0))])')
    if not has_contracts:
        injections.append('action_contracts: crate::shard::tests::test_fixtures::contracts_through(vb_core::ActionId::new(0))')
    if injections:
        stripped = new_inner.rstrip()
        if stripped and not stripped.endswith(','):
            new_inner = stripped + ',\n            ' + ',\n            '.join(injections)
        elif stripped:
            new_inner = stripped + '\n            ' + ',\n            '.join(injections)
        else:
            new_inner = '            ' + ',\n            '.join(injections)
    new_expr = 'ShardCommand::SubmitWithInputsAndContracts {' + new_inner + '}'
    return body[:expr_start] + new_expr + body[brace_end+1:]

def process_file(filepath):
    with open(filepath, 'r') as f:
        source = f.read()
    changed_funcs = []
    for func_name in WHITELIST:
        func_range = find_function_range(source, func_name)
        if not func_range:
            continue
        start, end = func_range
        body = source[start:end]
        original_body = body
        exprs = find_submit_exprs(body)
        if not exprs:
            continue
        exprs.sort(key=lambda x: x[0], reverse=True)
        for expr_start, variant, brace_start, brace_end in exprs:
            body = transform_expr(body, expr_start, variant, brace_start, brace_end)
        if body != original_body:
            source = source[:start] + body + source[end:]
            changed_funcs.append(func_name)
    if changed_funcs:
        with open(filepath, 'w') as f:
            f.write(source)
    return changed_funcs

def main():
    all_changed = []
    test_dir = os.path.join(WORKDIR, "crates/vb_runtime/src/shard/lifecycle_tests")
    for filepath in glob.glob(os.path.join(test_dir, "chunk_*.rs")):
        changed = process_file(filepath)
        all_changed.extend(changed)
    for func_name in all_changed:
        print(func_name)

if __name__ == '__main__':
    main()
