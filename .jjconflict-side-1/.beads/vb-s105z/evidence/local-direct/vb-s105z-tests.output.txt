#[test]
fn test_body_constant_index_basic() -> Result<(), Box<dyn std::error::Error>> {
    let mut compiler = SlotCompiler::new();
    let mut builder = compiler.new_builder();
    let idx1 = compiler.body_constant_index(&mut builder, "7", 1, true)?;
    assert_eq!(idx1, ConstIdx::new(0));
    let idx2 = compiler.body_constant_index(&mut builder, "99", 2, true)?;
    assert_eq!(idx2, ConstIdx::new(1));
    assert_eq!(builder.constants, vec![ConstValue::I64(7), ConstValue::I64(99)]);
    Ok(())
}

#[test]
fn test_body_constant_index_reuse() -> Result<(), Box<dyn std::error::Error>> {
    let mut compiler = SlotCompiler::new();
    let mut builder = compiler.new_builder();
    compiler.body_constant_index(&mut builder, "7", 1, true)?;
    compiler.body_constant_index(&mut builder, "99", 2, true)?;
    let idx = compiler.body_constant_index(&mut builder, "7", 3, true)?;
    assert_eq!(idx, ConstIdx::new(0));
    assert_eq!(builder.constants.len(), 2);
    Ok(())
}

#[test]
fn test_body_constant_index_parse_errors() -> Result<(), Box<dyn std::error::Error>> {
    let mut compiler = SlotCompiler::new();
    let mut builder = compiler.new_builder();
    compiler.body_constant_index(&mut builder, "1", 1, true)?;
    for bad in ["bad", "9223372036854775808", "-9223372036854775809"] {
        match compiler.body_constant_index(&mut builder, bad, 9, true) {
            Ok(_) => return Err("expected parse error".into()),
            Err(err) => assert!(err.contains(&CompileError::StepFieldShape {
                step: 9, field: "set.value".to_string(), expected: "integer string".to_string()
            })),
        }
    }
    assert_eq!(builder.constants.len(), 1);
    Ok(())
}

#[test]
fn test_body_constant_index_extremes() -> Result<(), Box<dyn std::error::Error>> {
    let mut compiler = SlotCompiler::new();
    let mut builder = compiler.new_builder();
    let idx_min = compiler.body_constant_index(&mut builder, &i64::MIN.to_string(), 1, true)?;
    let idx_max = compiler.body_constant_index(&mut builder, &i64::MAX.to_string(), 2, true)?;
    let idx_zero = compiler.body_constant_index(&mut builder, "0", 3, true)?;
    assert_eq!(idx_min, ConstIdx::new(0));
    assert_eq!(idx_max, ConstIdx::new(1));
    assert_eq!(idx_zero, ConstIdx::new(2));
    assert_eq!(builder.constants, vec![ConstValue::I64(i64::MIN), ConstValue::I64(i64::MAX), ConstValue::I64(0)]);
    Ok(())
}

#[test]
fn test_body_constant_index_overflow() -> Result<(), Box<dyn std::error::Error>> {
    let mut compiler = SlotCompiler::new();
    let mut builder = compiler.new_builder();
    builder.constants = vec![ConstValue::I64(0); 65536];
    match compiler.body_constant_index(&mut builder, "1", 1, true) {
        Ok(_) => return Err("expected overflow error".into()),
        Err(err) => assert!(err.contains(&CompileError::Workflow(
            WorkflowError::ConstOutOfBounds { constant: ConstIdx::new(u16::MAX) }
        ))),
    }
    assert_eq!(builder.constants.len(), 65536);
    Ok(())
}

#[test]
fn test_body_constant_index_reuse_when_full() -> Result<(), Box<dyn std::error::Error>> {
    let mut compiler = SlotCompiler::new();
    let mut builder = compiler.new_builder();
    builder.constants = vec![ConstValue::I64(0); 65536];
    let idx = compiler.body_constant_index(&mut builder, "0", 1, true)?;
    assert_eq!(idx, ConstIdx::new(0));
    Ok(())
}
