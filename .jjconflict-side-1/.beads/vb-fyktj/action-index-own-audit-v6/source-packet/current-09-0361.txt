401:     /// PO-012: After N successful staging operations, batch.len() ==
402:     /// num_appends * 2 (each action-scheduled event auto-stages the
403:     /// matching index_action mutation). `is_empty()` ==
404:     /// `(len() == 0)` always holds.
405:     ///
406:     /// vb-3wn7x: as of this fix, each `append_event(ActionScheduled)`
407:     /// appends the event AND its `index_action` mutation to the
408:     /// OwnedWriteBatch, so each successful `append_event` increments
409:     /// `batch.len()` by exactly 2 (event row + index marker). The
410:     /// total `batch.len()` after `N` appends is therefore `N * 2`,
411:     /// not `N`. The strict-monotonicity, `is_empty() == (len() == 0)`,
412:     /// and never-shrinks invariants remain.
413:     #[test]
414:     fn test_batch_len_monotonic_and_is_empty_invariant(
415:         num_ops in 0u8..=20u8,
416:         action_val in 1u16..=100u16,
417:         run_val in 1u64..=1000u64,
418:     ) {
419:         // vb-3wn7x: one for the event row + one for the auto-staged
420:         // index_action mutation. Future action-lifecycle variants that
421:         // stage additional index mutations would bump this constant.
422:         const LEN_INCREMENT_PER_APPEND_EVENT: usize = 2;
423: 
424:         let action = ActionId::new(action_val);
425:         let run = RunId::new(run_val);
426:         let (_temp, journal) = temp_journal();
427: 
428:         let mut batch = JournalWriteBatch::new(&journal);
429: 
430:         // Initial state invariants
431:         prop_assert_eq!(batch.len(), 0, "new batch len must be 0");
432:         prop_assert!(batch.is_empty(), "new batch must be empty");
433: 
434:         let num_ops = num_ops as usize;
435:         for i in 0..num_ops {
436:             let step = StepIdx::new(i as u16);
437:             let event = make_action_scheduled(run, i as u64, step, action);
438: 
439:             let before_len = batch.len();
440:             let _before_empty = batch.is_empty();
441: 
442:             let result = batch.append_event(&event);
443:             prop_assert!(result.is_ok(), "append_event must succeed for valid event");
444: 
445:             // After successful staging: len increases by the per-call
446:             // increment (event + auto-staged index_action).
447:             prop_assert_eq!(
448:                 batch.len(),
449:                 before_len + LEN_INCREMENT_PER_APPEND_EVENT,
450:                 "len must increment by 2 after successful staging (event + auto-staged index_action)",
451:             );
452:             prop_assert!(
453:                 batch.len() > before_len,
454:                 "len must be strictly greater after successful staging",
455:             );
456:             prop_assert!(
457:                 !batch.is_empty(),
458:                 "batch must not be empty after successful staging",
459:             );
460: 
461:             // is_empty == (len == 0) invariant must always hold
462:             prop_assert_eq!(
463:                 batch.is_empty(),
464:                 batch.len() == 0,
465:                 "is_empty() must equal (len() == 0) always",
466:             );
467:         }
468: 
469:         // After all staging: len == num_ops * 2 (each append_event
470:         // contributes the event row plus the auto-staged index_action
471:         // mutation).
472:         prop_assert_eq!(
473:             batch.len(),

479:             num_ops == 0,
