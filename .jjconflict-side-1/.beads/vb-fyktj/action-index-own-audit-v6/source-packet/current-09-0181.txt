260: // PO-009: Batch commit is all-or-nothing across keyspaces (proptest)
261: // ---------------------------------------------------------------------------
262: 
263: proptest::proptest! {
264:     #![proptest_config(journal_proptest_config(JOURNAL_IO_PROPTEST_CASES))]
265: 
266:     /// PO-009: When commit() returns Err, NO operations from the batch are visible
267:     /// in any keyspace.
268:     ///
269:     /// This test is inherently limited because Fjall's commit() rarely fails in practice.
270:     /// We verify the batch state machine behaves correctly by checking aborted flag
271:     /// and len() invariants after failed encode operations.
272:     ///
273:     /// vb-3wn7x: as of this fix, `append_event` for an action-lifecycle
274:     /// event automatically stages the corresponding `index_action`
275:     /// mutation in the same batch. A subsequent explicit
276:     /// `put_action_index` call from the test is therefore redundant —
277:     /// Fjall last-write-wins leaves the marker in place either way.
278:     /// The expected `batch.len()` after `append_event` of an
279:     /// `ActionScheduled` plus an explicit `put_action_index` is now 3:
280:     /// one for the event, one auto-staged by `append_event`, and one
281:     /// from the explicit call.
282:     #[test]
283:     fn test_batch_commit_all_or_nothing_across_keyspaces(
284:         action_val in 1u16..=100u16,
285:         run_val in 1u64..=1000u64,
286:         step_val in 0u16..=50u16,
287:     ) {
288:         let action = ActionId::new(action_val);
289:         let run = RunId::new(run_val);
290:         let step = StepIdx::new(step_val);
291:         let (_temp, journal) = temp_journal();
292: 
293:         let seq = EventSeq::new(0);
294:         let event = make_action_scheduled(run, seq.get(), step, action);
295: 
296:         let mut batch = JournalWriteBatch::new(&journal);
297: 
298:         // Stage valid entries. vb-3wn7x: append_event auto-stages the
299:         // index_action mutation, so it contributes 2 (event + index)
300:         // to batch.len(). The explicit put_action_index call adds one
301:         // more (Fjall last-write-wins makes it idempotent).
302:         prop_assert!(batch.append_event(&event).is_ok(), "append_event must succeed");
303:         prop_assert!(batch.put_action_index(action, run, step).is_ok(), "put_action_index must succeed");
304: 
305:         // Before commit: batch is non-empty
306:         prop_assert!(!batch.is_empty(), "batch must be non-empty before commit");
307:         prop_assert_eq!(
308:             batch.len(),
309:             3,
310:             "batch.len() must be 3 before commit (event + auto-index + explicit-index)"
311:         );
312: 
313:         // Commit succeeds with valid data — verifying all-or-nothing in practice
314:         let commit_result = batch.commit();
315:         prop_assert!(commit_result.is_ok(), "valid batch commit must succeed");
316: 
317:         // After successful commit: both entries are durable
318:         let events = journal.events_for_run(run).expect("events_for_run must succeed");
319:         prop_assert_eq!(events.len(), 1, "event must be durable after commit");
320: 
321:         let index_key = vb_storage::keys::index_action_key(action, run, step)
322:             .expect("index_action_key must succeed");
323:         let has_entry = journal.has_action_index_entry(index_key)
324:             .expect("has_action_index_entry must succeed");
325:         prop_assert!(has_entry, "index_action entry must be durable");
326:     }
327: }
