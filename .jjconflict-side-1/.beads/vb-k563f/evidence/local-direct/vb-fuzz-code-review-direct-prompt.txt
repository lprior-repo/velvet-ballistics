1: //! Fuzz target for IPC frame decode.
2: //!
3: //! ## INVARIANT Oracle
4: //!
5: //! Replaces crash-only fuzzing with structural assertions on
6: //! `decode_frame_header` and `decode_frame_payload`:
7: //! - Decoded `IpcFrameHeader` is guaranteed to have matched `IPC_VERSION` on
8: //!   decode (the version slot is validated during decode and not stored on the
9: //!   struct). The wire-format version slot of any re-encoded header MUST equal
10: //!   `IPC_VERSION`.
11: //! - On Ok: re-encoding the decoded header round-trips to the exact input
12: //!   bytes (`encode ∘ decode = id`).
13: //! - `decode_frame_payload` MUST return `Err` whenever `payload.len()` differs
14: //!   from the header's `payload_len` field.
15: //! - All errors are typed `IpcError` variants (enforced inside
16: //!   `fuzz_lib::fuzz_ipc_frame`).
17: //!
18: //! Every `decode_frame_header` call is matched into `Ok / Err` so that no
19: //! decoder `Result` is silently discarded and no wildcard `IpcPayload` arm is
20: //! reached. Decode and encode failures are routed through the typed
21: //! `fuzz_lib::assert_typed_ipc_error` oracle.
22: //!
23: //! Corpus seeds are maintained in `fuzz/corpus/ipc_frame/`.
24: 
25: #![no_main]
26: 
27: use libfuzzer_sys::fuzz_target;
28: 
29: fuzz_target!(|data: &[u8]| {
30:     // Target-level oracle: a successfully decoded header re-encodes to bytes
31:     // whose version slot equals IPC_VERSION. This is the only externally
32:     // observable form of the "decoded header has version == IPC_VERSION"
33:     // invariant, since `IpcFrameHeader` does not store a version field.
34:     //
35:     // Both decoder calls are matched into `Ok / Err` so the typed oracle in
36:     // `fuzz_lib` exercises every error class. The strict lint gate forbids
37:     // `let _ = decode(...)` discard patterns and any wildcard `IpcPayload`
38:     // arm, so we route both branches through the oracle.
39:     if let Some(header_bytes) = data.first_chunk::<{ vb_ipc::IPC_HEADER_LEN }>() {
40:         match vb_ipc::frame::decode_frame_header(header_bytes) {
41:             Ok(decoded) => match decoded.encode() {
42:                 Ok(encoded) => {
43:                     let version_bytes: [u8; 2] = encoded
44:                         .as_slice()
45:                         .get(4..6)
46:                         .and_then(|s| <[u8; 2]>::try_from(s).ok())
47:                         .unwrap_or([0u8; 2]);
48:                     let version = u16::from_le_bytes(version_bytes);
49:                     assert_eq!(
50:                         version,
51:                         vb_ipc::IPC_VERSION,
52:                         "decoded IpcFrameHeader re-encodes with version != IPC_VERSION"
53:                     );
54:                     assert_eq!(
55:                         encoded.as_slice(),
56:                         header_bytes.as_slice(),
57:                         "re-encoded header must match original bytes"
58:                     );
59:                 }
60:                 Err(encode_error) => {
61:                     assert!(fuzz_lib::assert_typed_ipc_error(encode_error));
62:                 }
63:             },
64:             Err(decode_error) => {
65:                 assert!(fuzz_lib::assert_typed_ipc_error(decode_error));
66:             }
67:         }
68:     }
69: 
70:     fuzz_lib::fuzz_ipc_frame(data);
71: });
