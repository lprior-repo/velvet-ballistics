583: impl Drop for InspectHandle {
584:     fn drop(&mut self) {
585:         if let Ok(mut guard) = self.registry.lock() {
586:             // Only remove if the epoch matches (handles stale drops correctly)
587:             if let Some(current_epoch) = guard.get(&self.run)
588:                 && *current_epoch == self.epoch
589:             {
590:                 guard.remove(&self.run);
591:             }
592:         }
593:     }

596: /// Registry for RAII-based introspection handles.
597: ///
598: /// Provides epoch-based registration with automatic cleanup on guard drop.
599: /// Does NOT create global mutable run state - each registry instance is independent.
600: #[derive(Default)]
601: pub struct IntrospectionRegistry {
602:     inner: std::sync::Arc<std::sync::Mutex<std::collections::HashMap<RunId, u64>>>,
603:     next_epoch: u64,
604: }
605: 
606: impl IntrospectionRegistry {
607:     /// Creates a new empty registry.
608:     #[must_use]
609:     pub fn new() -> Self {
610:         Self::default()
611:     }
612: 
613:     /// Acquires the registry's inner mutex, recovering from any prior
614:     /// poison state. The standard `Mutex::lock` returns `PoisonError`
615:     /// forever once a holder panics; this helper mirrors the recovery
616:     /// pattern already used by `action_queue` so the admission gate does
617:     /// not stay permanently disabled after a panic.
618:     fn lock_or_recover(
619:         inner: &std::sync::Arc<std::sync::Mutex<std::collections::HashMap<RunId, u64>>>,
620:     ) -> std::sync::MutexGuard<'_, std::collections::HashMap<RunId, u64>> {
621:         match inner.lock() {
622:             Ok(guard) => guard,
623:             Err(poisoned) => poisoned.into_inner(),
624:         }
625:     }
626: 
627:     /// Registers a handle for the given run.
628:     ///
629:     /// Returns the handle guard on success. Recovers transparently from
630:     /// a poisoned mutex so a single panicking holder cannot permanently
631:     /// disable the admission gate (RA-014).
632:     pub fn register(&mut self, run: RunId) -> RuntimeResult<InspectHandle> {
633:         let mut guard = Self::lock_or_recover(&self.inner);
634: 
635:         // Check if already registered
636:         if guard.contains_key(&run) {
637:             return Err(crate::RuntimeError::RunAlreadyExists);
638:         }
639: 
640:         let epoch = self.next_epoch;
641:         self.next_epoch = self
642:             .next_epoch
643:             .checked_add(1)
644:             .ok_or(crate::RuntimeError::IntrospectionEpochExhausted)?;
645:         guard.insert(run, epoch);
646: 
647:         Ok(InspectHandle {
648:             run,
649:             epoch,
650:             registry: self.inner.clone(),
651:         })
652:     }
653: 
654:     /// Registers a handle for the given run, allowing epoch replacement on conflict.
655:     ///
656:     /// Returns outcome indicating whether registration succeeded, conflicted, or was replaced.
657:     pub fn register_with_overlap_policy(
658:         &mut self,
659:         run: RunId,
660:     ) -> RuntimeResult<(InspectHandle, Result<(), RegisterOverlapOutcome>)> {
661:         let mut guard = Self::lock_or_recover(&self.inner);
662: 
663:         let (outcome, epoch) = if let Some(&old_epoch) = guard.get(&run) {
664:             // Overlap detected - replace with new epoch
665:             let new_epoch = self.next_epoch;
666:             self.next_epoch = self
667:                 .next_epoch
668:                 .checked_add(1)
669:                 .ok_or(crate::RuntimeError::IntrospectionEpochExhausted)?;
670:             guard.insert(run, new_epoch);
671:             (
672:                 Err(RegisterOverlapOutcome::Replaced {
673:                     old_epoch,
674:                     new_epoch,
675:                 }),
676:                 new_epoch,
677:             )
678:         } else {
679:             // No overlap - insert with new epoch
680:             let epoch = self.next_epoch;
681:             self.next_epoch = self
682:                 .next_epoch
683:                 .checked_add(1)
684:                 .ok_or(crate::RuntimeError::IntrospectionEpochExhausted)?;
685:             guard.insert(run, epoch);
686:             (Ok(()), epoch)
687:         };
688: 
689:         Ok((
690:             InspectHandle {
691:                 run,
692:                 epoch,
693:                 registry: self.inner.clone(),
694:             },
695:             outcome,
696:         ))
697:     }

700:     ///
701:     /// Returns whether the handle was found and unregistered.
702:     pub fn unregister(&mut self, run: RunId) -> RuntimeResult<UnregisterOutcome> {
703:         let mut guard = Self::lock_or_recover(&self.inner);
704: 
705:         if guard.remove(&run).is_some() {
706:             Ok(UnregisterOutcome::Unregistered)
707:         } else {
708:             Ok(UnregisterOutcome::Missing)
709:         }
710:     }

719:         Ok(count)

784: /// Domain-named wrapper around `crossbeam_queue::ArrayQueue<ShardCommand>`.
785: ///
786: /// Provides a bounded, non-blocking command queue with domain-specific terminology
787: /// (`enqueue`, `pop`, `is_full`, `remaining_capacity`) and proper error taxonomy
788: /// (`RuntimeError::QueueFull`). This wrapper establishes the `ShardCommand` queue
789: /// as a first-class domain boundary rather than a raw field.
790: pub struct ShardCommandQueue {
791:     inner: ArrayQueue<ShardCommand>,
792:     /// Stored capacity to satisfy POST-001 and INV-001 invariants.
793:     capacity: usize,
794: }
795: 
796: impl ShardCommandQueue {
797:     /// Creates a new `ShardCommandQueue` with the given capacity.
798:     ///
799:     /// # Errors
800:     /// Returns `RuntimeError::CommandQueueCapacityExceeded` if `capacity` is 0
801:     /// or exceeds `MAX_COMMAND_QUEUE_CAPACITY`.
802:     pub fn new(capacity: usize) -> RuntimeResult<Self> {
803:         if !is_valid_command_queue_capacity(capacity) {
804:             return Err(crate::RuntimeError::CommandQueueCapacityExceeded {
805:                 capacity,
806:                 max: MAX_COMMAND_QUEUE_CAPACITY,
807:             });
808:         }
809:         Ok(Self {
810:             inner: ArrayQueue::new(capacity),
811:             capacity,
812:         })
813:     }
814: 
815:     /// Creates a command queue from an already-accepted shard configuration.
816:     ///
817:     /// `Shard::new` has historically been infallible and accepted `ShardConfig`
818:     /// by value. The validated constructor for externally supplied capacity is
819:     /// `ShardConfig::new`; this helper preserves `Shard::new`'s existing shape
820:     /// while placing the raw queue construction behind the domain wrapper.
821:     pub(crate) fn from_config(config: ShardConfig) -> Self {
822:         Self {
823:             inner: ArrayQueue::new(config.command_queue_capacity),
824:             capacity: config.command_queue_capacity,
825:         }
826:     }
827: 
828:     /// Enqueues a command. Returns `Ok(())` if the command was enqueued, or
829:     /// `Err(RuntimeError::QueueFull)` if the queue is at capacity.
830:     ///
831:     /// This operation is non-blocking and never allocates on failure.
832:     pub fn enqueue(&self, cmd: ShardCommand) -> RuntimeResult<()> {
833:         self.inner
834:             .push(cmd)
835:             .map_err(|_| crate::RuntimeError::QueueFull)
836:     }
837: 
838:     /// Dequeues the frontmost command, if any.
839:     ///
840:     /// Returns `Some(cmd)` in FIFO order, or `None` if the queue is empty.
841:     pub fn pop(&self) -> Option<ShardCommand> {
842:         self.inner.pop()
843:     }
844: 
845:     /// Returns the number of commands currently in the queue.
846:     #[must_use]
847:     pub fn len(&self) -> usize {
848:         self.inner.len()
849:     }
850: 
851:     /// Returns `true` if the queue contains no commands.
852:     #[must_use]
853:     pub fn is_empty(&self) -> bool {
854:         self.inner.is_empty()
855:     }
856: 
857:     /// Returns the capacity of this queue (set at construction).
858:     #[must_use]
859:     pub fn capacity(&self) -> usize {
860:         self.capacity
861:     }
862: 
863:     /// Returns the number of remaining free slots in the queue.
864:     #[must_use]
865:     pub fn remaining_capacity(&self) -> usize {
866:         self.capacity.saturating_sub(self.inner.len())
867:     }
868: 
869:     /// Returns `true` if the queue is at capacity.
870:     #[must_use]
871:     pub fn is_full(&self) -> bool {
872:         self.inner.len() == self.capacity
873:     }
874: 
875:     /// Returns the compile-time bounded capacity limit (65536).
876:     ///
877:     /// This is the maximum capacity any `ShardCommandQueue` can be configured with.
878:     #[must_use]
879:     pub const fn bounded_capacity() -> usize {
880:         MAX_COMMAND_QUEUE_CAPACITY
881:     }

1439: /// A monotonically increasing timer tick value, counting logical time units.
1440: ///
1441: /// Wraps a `u64` to provide type safety and checked arithmetic for deterministic
1442: /// clock control. One tick represents one logical time unit in the deterministic
1443: /// timer seam, operating alongside the existing wall-clock `Instant`-based timers.
1444: #[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
1445: pub struct TimerTick(u64);
1446: 
1447: impl TimerTick {
1448:     /// Creates a new timer tick at the given value.
1449:     #[must_use]
1450:     pub const fn new(value: u64) -> Self {
1451:         Self(value)
1452:     }
1453: 
1454:     /// Returns the inner `u64` value.
1455:     #[must_use]
1456:     pub const fn get(self) -> u64 {
1457:         self.0
1458:     }
1459: 
1460:     /// Advances the tick by a duration, returning the resulting tick.
1461:     ///
1462:     /// Returns `None` on overflow.
1463:     #[must_use]
1464:     pub fn checked_add(self, duration: TimerDuration) -> Option<Self> {
1465:         self.0.checked_add(duration.get()).map(Self)
1466:     }
1467: 
1468:     /// Returns `true` if this tick is at or past the given deadline.
1469:     #[must_use]
1470:     pub fn has_elapsed(self, deadline: TimerDeadline) -> bool {
1471:         self.0 >= deadline.get()
1472:     }
1473: }
1474: 
1475: /// A timer duration measured in ticks, representing a span of logical time.
1476: #[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
1477: pub struct TimerDuration(u64);
1478: 
1479: impl TimerDuration {
1480:     /// Creates a new duration with the given number of ticks.
1481:     #[must_use]
1482:     pub const fn new(ticks: u64) -> Self {
1483:         Self(ticks)
1484:     }
1485: 
1486:     /// Returns the inner `u64` value.
1487:     #[must_use]
1488:     pub const fn get(self) -> u64 {
1489:         self.0
1490:     }
1491: 
1492:     /// Returns the duration as a tick count.
1493:     #[must_use]
1494:     pub const fn as_ticks(self) -> u64 {
1495:         self.0
1496:     }
1497: 
1498:     /// Returns a zero-length duration.
1499:     #[must_use]
1500:     pub const fn zero() -> Self {
1501:         Self(0)
1502:     }
1503: }

1505: /// An absolute deadline in ticks, representing when a timer expires.
1506: #[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
1507: pub struct TimerDeadline(u64);

1509: impl TimerDeadline {
1510:     /// Creates a new deadline at the given tick value.
1511:     #[must_use]
1512:     pub const fn new(tick: u64) -> Self {
1513:         Self(tick)
1514:     }
1515: 
1516:     /// Returns the inner `u64` value.
1517:     #[must_use]
1518:     pub const fn get(self) -> u64 {
1519:         self.0
1520:     }
1521: 
1522:     /// Creates a deadline by adding a duration to a tick.
1523:     ///
1524:     /// Returns `None` on overflow.
1525:     #[must_use]
1526:     pub fn from_tick_and_duration(tick: TimerTick, duration: TimerDuration) -> Option<Self> {
1527:         tick.get().checked_add(duration.get()).map(Self)

2333: // Regression tests for RA-014: lock_admission permanently bricks shard
2334: // admission on mutex poison.
2335: //
2336: // Before the fix, `IntrospectionRegistry::register` / `unregister` /
2337: // `unregister_all` / `register_with_overlap_policy` all used the pattern
2338: //   `self.inner.lock().map_err(|_| RuntimeError::JournalPoisoned)?`
2339: // which discards the poisoned guard. Once poisoned, every subsequent call
2340: // hits a still-poisoned mutex and returns `Err(JournalPoisoned)` forever,
2341: // permanently disabling the admission gate for that shard.
2342: //
2343: // After the fix, the registry recovers the inner data via

2355:     /// Spawns a thread that locks the shared mutex and panics, leaving the
2356:     /// mutex poisoned. `catch_unwind` swallows the panic so the test thread
2357:     /// is not poisoned. The function returns once the spawned thread has
2358:     /// finished and the mutex is verifiably poisoned.
2359:     fn poison_arc_mutex<T: Send + 'static>(mutex: Arc<std::sync::Mutex<T>>) {

2362:             // itself is the source of the poison; we use `catch_unwind` to
2363:             // keep the test process alive.

2366:         });

2368:             // Wait for the spawner to finish panicking.
2369:             let _ = handle.join();
2370:         }));
2371:         // The spawned thread panicked; that is the intended outcome.
2372:         let _ = result;
