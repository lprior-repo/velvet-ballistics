46:     fn append_storage_event(&self, event: &JournalEvent) -> RuntimeResult<()> {
47:         let result = if self.profile == DurabilityProfile::Strict {
48:             self.journal.append_strict(event)
49:         } else {
50:             self.journal.append_journaled(event)
51:         };
52:         result.map_err(RuntimeError::from)
53:     }

55:     fn append_runtime_event_batch(
56:         journal: &FjallJournal,
57:         profile: DurabilityProfile,
58:         events: &[RuntimeJournalEvent],
59:         start_seq: EventSeq,
60:     ) -> RuntimeResult<()> {
61:         if events.is_empty() {
62:             return Ok(());
63:         }
64:         let mut batch = journal.batch();
65:         for (offset, event) in events.iter().enumerate() {
66:             let seq = Self::sequence_at_offset(start_seq, offset)?;
67:             let storage_event = Self::storage_event(event.clone(), seq)?;
68:             batch
69:                 .append_event(&storage_event)
70:                 .map_err(RuntimeError::from)?;
71:         }
72:         if profile == DurabilityProfile::Strict {
73:             return batch.strict().commit().map_err(RuntimeError::from);
74:         }
75:         batch.commit().map_err(RuntimeError::from)
76:     }

78:     fn sequence_at_offset(start_seq: EventSeq, offset: usize) -> RuntimeResult<EventSeq> {
79:         let offset = u64::try_from(offset)
80:             .map_err(|_| RuntimeError::from(vb_storage::JournalError::SequenceOverflow))?;
81:         start_seq
82:             .get()
83:             .checked_add(offset)
84:             .map(EventSeq::new)
85:             .ok_or_else(|| RuntimeError::from(vb_storage::JournalError::SequenceOverflow))
86:     }

88:     fn run_storage_event(
89:         event: RuntimeJournalEvent,
90:         seq: EventSeq,
91:     ) -> RuntimeResult<Option<JournalEvent>> {
92:         match event {
93:             RuntimeJournalEvent::RunSubmitted { run, workflow } => {
94:                 Ok(Some(JournalEvent::RunAccepted { run, seq, workflow }))
95:             }
96:             RuntimeJournalEvent::RunAdmission { admission } => {
97:                 Ok(Some(JournalEvent::RunAdmission {
98:                     input_digest: vb_storage::RunInputBinding::LegacyUnbound,
99:                     run: admission.run_id(),
100:                     seq,
101:                     artifact_digest: admission.artifact_digest(),
102:                     policy_pin: admission.policy_pin(),
103:                     granted_capabilities: admission.granted_capabilities().clone(),
104:                     policy: admission.policy(),
105:                 }))
106:             }
107:             RuntimeJournalEvent::RetryCheckpoint {
108:                 run,
109:                 attempt,
110:                 step,
111:                 payload_digest,
112:                 payload,
113:             } => Ok(Some(JournalEvent::RetryCheckpoint {
114:                 run,
115:                 seq,
116:                 attempt,
117:                 step,
118:                 payload_digest,
119:                 payload,
120:             })),
121:             RuntimeJournalEvent::RunFinished {
122:                 run,
123:                 result,
124:                 attempt,
125:             } => Ok(Some(JournalEvent::RunFinished {
126:                 run,
127:                 seq,
128:                 result,
129:                 attempt,
130:             })),
131:             RuntimeJournalEvent::RunFailed { run, attempt } => {
132:                 Ok(Some(JournalEvent::RunFailedEvent { run, seq, attempt }))
133:             }
134:             RuntimeJournalEvent::RunCancelled {
135:                 run,
136:                 reason,
137:                 attempt,
138:             } => Ok(Some(JournalEvent::RunCancelled {
139:                 run,
140:                 seq,
141:                 attempt,
142:                 reason,
143:             })),
144:             RuntimeJournalEvent::RunKilled { run, attempt } => {
145:                 Ok(Some(JournalEvent::RunKilled { run, seq, attempt }))
146:             }
147:             RuntimeJournalEvent::Resumed { run, timestamp } => {
148:                 resumed_storage_event(run, seq, timestamp).map(Some)
149:             }
150:             RuntimeJournalEvent::StepStarted { run, step, attempt } => {
151:                 Ok(Some(JournalEvent::StepStarted {
152:                     run,
153:                     seq,
154:                     step,
155:                     attempt,
156:                 }))
157:             }
158:             RuntimeJournalEvent::StepSucceeded {
159:                 run, step, output, ..
160:             } => Ok(Some(JournalEvent::StepSucceeded {
161:                 run,
162:                 seq,
163:                 step,
164:                 output,
165:             })),
166:             RuntimeJournalEvent::ActionScheduled { .. }
167:             | RuntimeJournalEvent::ActionCompleted { .. }
168:             | RuntimeJournalEvent::ActionScheduledTicket { .. }

350:     fn storage_event(event: RuntimeJournalEvent, seq: EventSeq) -> RuntimeResult<JournalEvent> {
351:         // Dispatch on `&event` so the full event is cloned at most once, by the
352:         // matched arm that actually consumes it. The previous chain cloned the
353:         // event three times before the real variant was determined.
354:         let result = match &event {
355:             RuntimeJournalEvent::RunSubmitted { .. }
356:             | RuntimeJournalEvent::RunAdmission { .. }
357:             | RuntimeJournalEvent::RetryCheckpoint { .. }
358:             | RuntimeJournalEvent::RunFinished { .. }
359:             | RuntimeJournalEvent::RunFailed { .. }
360:             | RuntimeJournalEvent::RunCancelled { .. }
361:             | RuntimeJournalEvent::RunKilled { .. }
362:             | RuntimeJournalEvent::Resumed { .. }
363:             | RuntimeJournalEvent::StepStarted { .. }
364:             | RuntimeJournalEvent::StepSucceeded { .. } => {
365:                 Self::run_storage_event(clone_for_dispatch(&event), seq)
366:             }
367:             RuntimeJournalEvent::ActionScheduled { .. }
368:             | RuntimeJournalEvent::ActionCompleted { .. }
369:             | RuntimeJournalEvent::ActionScheduledTicket { .. }
370:             | RuntimeJournalEvent::ActionCompletedEnvelope { .. }
371:             | RuntimeJournalEvent::ActionFailed { .. }
372:             | RuntimeJournalEvent::ActionAbandoned { .. } => {
373:                 Ok(Self::action_storage_event(clone_for_dispatch(&event), seq))
374:             }
375:             RuntimeJournalEvent::WaitScheduled { .. }
376:             | RuntimeJournalEvent::WaitResolved { .. }
377:             | RuntimeJournalEvent::AskScheduled { .. }
378:             | RuntimeJournalEvent::AskAnswered { .. }
379:             | RuntimeJournalEvent::AskTimedOut { .. }
380:             | RuntimeJournalEvent::SlotWritten { .. } => {
381:                 Self::boundary_storage_event(clone_for_dispatch(&event), seq)
382:             }
383:         }?;
384:         if let Some(storage_event) = result {
385:             return Ok(storage_event);
386:         }
387:         Err(RuntimeError::UnsupportedRuntimeJournalEventMapping {
388:             event_kind: event.kind(),
389:         })
390:     }
391: }

393: fn resumed_storage_event(run: RunId, seq: EventSeq, timestamp: u64) -> RuntimeResult<JournalEvent> {
394:     let seconds =
395:         i64::try_from(timestamp).map_err(|_| RuntimeError::RuntimeJournalTimestampOutOfRange {
396:             event_kind: "Resumed",
397:             timestamp,
398:         })?;
399:     let timestamp = DateTime::<Utc>::from_timestamp(seconds, 0).ok_or(
400:         RuntimeError::RuntimeJournalTimestampOutOfRange {
401:             event_kind: "Resumed",
402:             timestamp,
403:         },
404:     )?;
405:     Ok(JournalEvent::RunResumed {
406:         run,
407:         seq,
408:         timestamp,
409:     })
410: }

412: /// Counts the number of times `clone_for_dispatch` is invoked. Used by the
413: /// `storage_event_clones_the_event_exactly_once_per_dispatch` regression test
414: /// to assert that dispatching a `RuntimeJournalEvent` through `storage_event`
415: /// performs exactly one full-event clone.
416: #[cfg(test)]
417: static STORAGE_EVENT_CLONE_COUNT: std::sync::atomic::AtomicUsize =
418:     std::sync::atomic::AtomicUsize::new(0);
419: 
420: /// Clones a `RuntimeJournalEvent` for dispatch into one of the per-variant
421: /// helpers. In test builds this also increments `STORAGE_EVENT_CLONE_COUNT`
422: /// so the single-clone invariant is enforced by an executable assertion; in
423: /// release builds it is a plain clone.
424: fn clone_for_dispatch(event: &RuntimeJournalEvent) -> RuntimeJournalEvent {
425:     #[cfg(test)]
426:     {
427:         STORAGE_EVENT_CLONE_COUNT.fetch_add(1, std::sync::atomic::Ordering::SeqCst);
428:     }
429:     event.clone()
430: }
