45:     pub(super) fn require_pending_action_ownership(
46:         &self,
47:         ticket: ActionTicket,
48:     ) -> RuntimeResult<()> {
49:         if !self.run_state_contains(ticket.run) {
50:             return Err(RuntimeError::RunNotFound);
51:         }
52:         match self.pending_action_get(ticket.run) {
53:             Some(pending) if pending == ticket => Ok(()),
54:             Some(pending)
55:                 if ticket.attempt > 0
56:                     && ticket.attempt < pending.attempt
57:                     && ticket.run == pending.run
58:                     && ticket.step == pending.step
59:                     && ticket.action == pending.action
60:                     && ticket.capacity == pending.capacity
61:                     && vb_core::action::action_ticket_has_valid_key(ticket)
62:                     && ticket.seq <= pending.seq =>
63:             {
64:                 Err(RuntimeError::StaleAttempt {
65:                     incoming: ticket.attempt,
66:                     current: pending.attempt,
67:                 })
68:             }
69:             Some(_) | None => Err(RuntimeError::InvalidActionCompletion),
70:         }
71:     }
