1: //! YAML parsing fuzzing targets.
2: //
3: // The strict fuzz clippy denies `indexing_slicing`, `as_conversions`,
4: // `let_underscore_must_use`, and `arithmetic_side_effects`. The broad
5: // `#![allow(...)]` lines that previously suppressed those lints have been
6: // removed so the strict gate is enforceable. The remaining allows are
7: // documentary lints the strict command does not deny.
8: #![allow(clippy::missing_errors_doc)]
9: #![allow(clippy::must_use_candidate)]
10: #![allow(clippy::missing_panics_doc)]
11: #![allow(clippy::too_many_lines)]
12: #![allow(clippy::items_after_statements)]
13: #![allow(clippy::doc_markdown)]
14: #![allow(clippy::len_zero)]
15: 
16: const MAX_FUZZ_PAYLOAD_USIZE: usize = 100_000;
17: 
18: fn assert_typed_yaml_error(error: vb_compile::YamlError) {
19:     use vb_compile::YamlError;
20:     match error {
21:         YamlError::UnsupportedTrigger { .. }
22:         | YamlError::UnsupportedFeature { .. }
23:         | YamlError::DuplicateKey { .. }
24:         | YamlError::AnchorAliasMerge
25:         | YamlError::CustomTag { .. }
26:         | YamlError::BinaryScalar
27:         | YamlError::MultipleDocuments { .. }
28:         | YamlError::AmbiguousScalar { .. }
29:         | YamlError::SourceTooLarge { .. }
30:         | YamlError::NestingTooDeep { .. }
31:         | YamlError::NodeLimitExceeded { .. }
32:         | YamlError::ScalarTooLong { .. }
33:         | YamlError::SequenceTooLong { .. }
34:         | YamlError::MappingTooLarge { .. }
35:         | YamlError::UnknownField { .. }
36:         | YamlError::EmptySource
37:         | YamlError::MissingField { .. }
38:         | YamlError::FieldShape { .. }
39:         | YamlError::ParseError { .. }
40:         | YamlError::ForbiddenFeature { .. }
41:         | YamlError::LegacyPrimitive { .. } => {}
42:         _ => {}
43:     }
44: }

46: pub fn fuzz_yaml_events(data: &[u8]) {
47:     if let Ok(text) = std::str::from_utf8(data) {
48:         let profile_result = vb_compile::validate_yaml_profile(text);
49:         match profile_result {
50:             Ok(()) => {}
51:             Err(e) => {
52:                 assert_typed_yaml_error(e);
53:             }
54:         }
55: 
56:         let events_result = vb_compile::parse_yaml_events(text);
57:         match events_result {
58:             Ok(events) => {
59:                 if !text.trim().is_empty() {
60:                     assert!(
61:                         !events.is_empty(),
62:                         "non-empty YAML input must produce non-empty events"
63:                     );
64:                 }
65:                 assert!(
66:                     events.len() <= MAX_FUZZ_PAYLOAD_USIZE,
67:                     "event count {} exceeds max payload bound",
68:                     events.len()
69:                 );
70:             }
71:             Err(e) => {
72:                 assert_typed_yaml_error(e);
73:             }
74:         }
75: 
76:         let source_map_result = vb_compile::build_source_map(text);
77:         match source_map_result {
78:             Ok(source_map) => {
79:                 assert!(
80:                     source_map.len() <= MAX_FUZZ_PAYLOAD_USIZE,
81:                     "source map entries {} exceeds max payload bound",
82:                     source_map.len()
83:                 );
84:             }
85:             Err(e) => {
86:                 assert_typed_yaml_error(e);
87:             }
88:         }
89:     }
90: }

92: pub fn fuzz_strict_yaml_profile(data: &[u8]) {
93:     let compile_result = vb_compile::compile_workflow(data);
94:     if let Ok(ref workflow) = compile_result {
95:         let text = String::from_utf8_lossy(data);
96:         let unsupported =
97:             text.contains("---") || text.contains('&') || text.contains('*') || text.contains('!');
98:         assert!(
99:             !unsupported,
100:             "unsupported YAML features must cause compile error"
101:         );
102:         assert!(
103:             workflow.node_count() >= 1,
104:             "compiled workflow must have at least 1 node"
105:         );
106:     }
107: }

109: pub fn fuzz_compile_source_ast_marks(data: &[u8]) {
110:     use vb_compile::compile_workflow;
111: 
112:     let result = compile_workflow(data);
113: 
114:     match result {
115:         Ok(_compiled) => {}
116:         Err(errors) => {
117:             assert!(
118:                 !errors.is_empty(),
119:                 "CompileErrors must contain at least one error"
120:             );
121:         }
122:     }
123: }

125: pub fn fuzz_span_bridge(data: &[u8]) {
126:     use vb_compile::{SourceSpan, build_semantic_source_map, build_source_map};
127: 
128:     let mut values = [0usize; 6];
129:     for (slot, byte) in values.iter_mut().zip(data.iter().copied()) {
130:         *slot = usize::from(byte);
131:     }
132: 
133:     let span = SourceSpan::new(
134:         values[0], values[1], values[2], values[3], values[4], values[5],
135:     );
136:     assert_eq!(span.start_offset, values[0]);
137:     assert_eq!(span.end_offset, values[1]);
138:     assert_eq!(span.start_line, values[2]);
139:     assert_eq!(span.start_col, values[3]);
140:     assert_eq!(span.end_line, values[4]);
141:     assert_eq!(span.end_col, values[5]);
142: 
143:     let Ok(text) = std::str::from_utf8(data) else {
144:         return;
145:     };
146: 
147:     if let Ok(map) = build_source_map(text) {
148:         for (index, mapped_span) in map.iter() {
149:             assert_eq!(map.span_for_node(index), Some(mapped_span));
150:         }
151:     }
152: 
153:     if let Ok(map) = build_semantic_source_map(text) {
154:         let _root = map.span_for_path("$");
155:         let _manual = map.span_for_path("$.when.manual");
156:         let _step_zero = map.span_for_path("$.steps[0]");
157:         // `_root`, `_manual`, and `_step_zero` are deliberately bound to
158:         // underscore-prefixed identifiers so the strict
159:         // `let_underscore_must_use` lint stays triggered only on
160:         // bare `let _ = must_use_call()`. They prove the lookup table
161:         // returns a span option without exercising it.
162:     }
163: }

165: #[cfg(test)]
166: mod tests {
167:     use super::*;
168: 
169:     #[test]
170:     fn large_yaml_exceeds_4096_bound() -> Result<(), Box<dyn std::error::Error>> {
171:         // Build valid YAML with >4096 events: a sequence of 5001 scalars
172:         // stays within default max_sequence_len=10000 and max_source_bytes=1048576.
173:         let item_count = 5001usize;
174:         let mut yaml = "items:\n".to_string();
175:         for i in 0..item_count {
176:             yaml.push_str(&format!("  - item{}\n", i));
177:         }
178: 
179:         // Validate via the real compile pipeline
180:         vb_compile::validate_yaml_profile(&yaml)?;
181: 
182:         let events = vb_compile::parse_yaml_events(&yaml)?;
183:         assert!(
184:             events.len() > 4096,
185:             "test YAML must produce >4096 events, got {}",
186:             events.len()
187:         );
188:         assert!(
189:             events.len() <= 100000,
190:             "test YAML must stay within default max_nodes, got {}",
191:             events.len()
192:         );
193: 
194:         let source_map = vb_compile::build_source_map(&yaml)?;
195:         assert!(
196:             source_map.len() > 4096,
197:             "test YAML must produce >4096 source map entries, got {}",
198:             source_map.len()
199:         );
200:         assert!(
201:             source_map.len() <= 100000,
202:             "test YAML must stay within default max_nodes, got {}",
203:             source_map.len()
204:         );
205: 
206:         // This assertion will fail with MAX_FUZZ_PAYLOAD_USIZE=4096 (RED)
207:         // and pass once the constant is fixed to 100000 (GREEN).
208:         fuzz_yaml_events(yaml.as_bytes());
209:         Ok(())
210:     }
