use super::{map_runtime_inputs, InputMappingError};
use crate::app_impl::prelude::*;

fn fixture_with_input_slots(input_slots: Vec<CompiledInputSlot>) -> Result<CompiledWorkflow, Box<dyn std::error::Error>> {
    let node = CompiledNode {
        id: 0,
        kind: CompiledNodeKind::Finish { result: 2 },
        targets: Box::from([]),
        exprs: Box::from([]),
        accessors: Box::from([]),
        constants: Box::from([]),
    };
    let parts = WorkflowParts {
        name: Box::from("test"),
        digest: WorkflowDigest([0; 32]),
        nodes: Box::from([node]),
        expressions: Box::from([]),
        accessors: Box::from([]),
        constants: Box::from([]),
        slot_count: 4,
        symbols_count: 0,
        entry: 0,
        resource_contract: ResourceContract::default(),
        step_names: Box::from([]),
        input_slots: Box::from(input_slots),
    };
    match CompiledWorkflow::try_from_parts(parts) {
        Ok(c) => Ok(c),
        Err(e) => Err(Box::from(e.to_string())),
    }
}

fn serialize_slot_values(values: Vec<SlotValue>) -> Result<Vec<u8>, Box<dyn std::error::Error>> {
    let boxed = Box::<[SlotValue]>::from(values);
    let bytes = postcard::to_allocvec(&boxed)?;
    Ok(bytes)
}

fn slot_value_i64(n: i64) -> SlotValue {
    SlotValue::I64(n)
}

fn slot_value_bool(b: bool) -> SlotValue {
    SlotValue::Bool(b)
}
