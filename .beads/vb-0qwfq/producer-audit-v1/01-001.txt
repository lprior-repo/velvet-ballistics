53: /// Recompute checks and evidence from hostile artifact bytes before runtime use.
54: /// Legacy claims require regeneration; a rehashed forged ledger is still rejected.
55: pub fn verify_accepted_artifact(
56:     artifact: &AcceptedArtifact,
57: ) -> Result<CompiledWorkflow, JournalError> {
58:     let len = u32::try_from(artifact.ir.len()).map_err(|_| JournalError::ArtifactMalformed)?;
59:     if len > crate::constants::MAX_COMPILED_IR_BYTES {
60:         return Err(JournalError::PayloadTooLarge {
61:             len,
62:             max: crate::constants::MAX_COMPILED_IR_BYTES,
63:         });
64:     }
65:     let evidence = checked_evidence_schema(&artifact.verification.evidence)?;
66:     if accepted_artifact_digest(artifact)? != artifact.digest
67:         || artifact.verification.digest != artifact.digest
68:     {
69:         return Err(JournalError::ArtifactChecksumMismatch);
70:     }
71:     let (parts, trailing) = postcard::take_from_bytes::<WorkflowParts>(&artifact.ir)
72:         .map_err(JournalError::PostcardDecodeFailed)?;
73:     if !trailing.is_empty() {
74:         return Err(JournalError::ArtifactMalformed);
75:     }
76:     let workflow =
77:         CompiledWorkflow::try_from_parts(parts).map_err(JournalError::WorkflowReconstruction)?;
78:     let expected = prepare_compiled_artifact(
79:         &workflow,
80:         evidence.admission_policy,
81:         &artifact.action_contracts,
82:     )?;
83:     verify_expected_bindings(artifact, &expected)?;
84:     Ok(workflow)
85: }
