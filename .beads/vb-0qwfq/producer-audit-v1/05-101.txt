155:     fn load_accepted_artifact(
156:         &self,
157:         artifact_digest: WorkflowDigest,
158:     ) -> Result<vb_storage::admission::AcceptedArtifact, ArtifactEnvelopeError> {
159:         // Load the compiled IR record from the journal. Accepted artifacts are
160:         // stored by artifact digest, while runtime submissions commonly carry
161:         // the workflow/source digest. Preserve the content-addressed key and
162:         // fall back to a bounded source-digest lookup for that runtime path.
163:         let record = match self.journal.compiled_ir(artifact_digest) {
164:             Ok(Some(record)) => record,
165:             Ok(None) => self
166:                 .journal
167:                 .compiled_ir_for_source_digest(artifact_digest)
168:                 .map_err(|_jb_err| ArtifactEnvelopeError::ArtifactNotFound {
169:                     digest: artifact_digest,
170:                 })?
171:                 .ok_or(ArtifactEnvelopeError::ArtifactNotFound {
172:                     digest: artifact_digest,
173:                 })?,
174:             Err(_) => {
175:                 return Err(ArtifactEnvelopeError::ArtifactNotFound {
176:                     digest: artifact_digest,
177:                 });
178:             }
179:         };
180: 
181:         // Decode the postcard payload as AcceptedArtifact.
182:         let artifact: vb_storage::admission::AcceptedArtifact = postcard::from_bytes(&record.ir)
183:             .map_err(|_decode_err| ArtifactEnvelopeError::PostcardDecodeFailed)?;
184: 
185:         if artifact.digest != artifact_digest && artifact.source_digest != artifact_digest {
186:             return Err(ArtifactEnvelopeError::ArtifactDigestMismatch {
187:                 requested: artifact_digest,
188:                 found: artifact.digest,
189:             });
190:         }
191:         validate_accepted_artifact_envelope(self, &artifact)?;
192: 
193:         Ok(artifact)
194:     }
195: }
