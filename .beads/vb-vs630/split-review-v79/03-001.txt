27: fn check_malformed_counter(bytes: &[u8], expected_actual_len: usize) -> TestResult {
28:     let (_dir, journal) = open_test_journal()?;
29:     write_malformed_counter(&journal, bytes)?;
30:     // Verify the raw malformed counter is stored as-is (don't call production read_counter,
31:     // which returns None on length mismatch and would exit early)
32:     let before = journal
33:         .artifact_acceptance
34:         .get(b"last_issued")?
35:         .ok_or("counter row missing")?;
36:     assert_eq!(before.as_ref(), bytes);
37: 
38:     let submission = make_submission(RuntimePolicy::Strict, 99)?;
39:     let result = journal.commit_run_submission(&submission, RuntimePolicy::Strict);
40:     match result {
41:         Err(JournalError::MalformedKeyspaceRow {
42:             prefix,
43:             expected_len,
44:             actual_len,
45:         }) => {
46:             assert_eq!(prefix, 0);
47:             assert_eq!(expected_len, 8);
48:             assert_eq!(actual_len, expected_actual_len);
49:         }
50:         other => return Err(format!("expected MalformedKeyspaceRow, got: {other:?}").into()),
51:     }
52:     // Verify counter unchanged and journal otherwise empty
53:     let after = journal
54:         .artifact_acceptance
55:         .get(b"last_issued")?
56:         .ok_or("counter row missing")?;
57:     assert_eq!(after.as_ref(), before.as_ref());
58:     verify_empty_journal(&journal)?;
59:     Ok(())
60: }
