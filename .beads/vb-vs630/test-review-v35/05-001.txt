25: fn check_malformed_counter(bytes: &[u8], expected_actual_len: usize) -> TestResult {
26:     let (_dir, journal) = open_test_journal()?;
27:     write_malformed_counter(&journal, bytes)?;
28:     // Verify the raw malformed counter is stored as-is (don't call production read_counter,
29:     // which returns None on length mismatch and would exit early)
30:     let before = journal
31:         .artifact_acceptance
32:         .get(b"last_issued")?
33:         .ok_or("counter row missing")?;
34:     assert_eq!(before.as_ref(), bytes);
35: 
36:     let submission = make_submission(RuntimePolicy::Strict, 99)?;
37:     let result = journal.commit_run_submission(&submission, RuntimePolicy::Strict);
38:     match result {
39:         Err(JournalError::MalformedKeyspaceRow {
40:             prefix,
41:             expected_len,
42:             actual_len,
43:         }) => {
44:             assert_eq!(prefix, 0);
45:             assert_eq!(expected_len, 8);
46:             assert_eq!(actual_len, expected_actual_len);
47:         }
48:         other => return Err(format!("expected MalformedKeyspaceRow, got: {other:?}").into()),
49:     }
50:     // Verify counter unchanged and journal otherwise empty
51:     let after = journal
52:         .artifact_acceptance
53:         .get(b"last_issued")?
54:         .ok_or("counter row missing")?;
55:     assert_eq!(after.as_ref(), before.as_ref());
56:     verify_empty_journal(&journal)?;
57:     Ok(())
58: }

60: /// Counter at u64::MAX yields exact SequenceOverflow error.
61: #[test]
62: fn cli_counter_at_max_yields_sequence_overflow() -> TestResult {
63:     let (_dir, journal) = open_test_journal()?;
64:     write_counter(&journal, u64::MAX)?;
65: 
66:     let submission = make_submission(RuntimePolicy::Strict, 1)?;
67:     let result = journal.commit_run_submission(&submission, RuntimePolicy::Strict);
68:     match result {
69:         Err(JournalError::SequenceOverflow) => {}
70:         other => return Err(format!("expected SequenceOverflow, got: {other:?}").into()),
71:     }
72:     assert_eq!(read_counter(&journal)?, Some(u64::MAX));
73:     Ok(())
74: }
