20: /// Creates a minimal AcceptedArtifact fixture suitable for prepare_artifact_with_sequence.

22: /// bind_artifact_digest after construction.
23: pub(super) fn minimal_artifact() -> TestResult {
24:     let proof = crate::admission::VerificationProof::new(
25:         WorkflowDigest::from_bytes([0u8; 32]),
26:         0,
27:         false,
28:     );
29:     let mut artifact = AcceptedArtifact {
30:         digest: WorkflowDigest::from_bytes([0u8; 32]),
31:         source_digest: WorkflowDigest::from_bytes([1u8; 32]),
32:         policy_digest: WorkflowDigest::from_bytes([2u8; 32]),
33:         ir: Vec::new(),
34:         verification: proof,
35:         accepted_at_seq: EventSeq::new(0),
36:         required_capabilities: Box::new([]),
37:         action_contracts: Box::new([]),
38:     };
