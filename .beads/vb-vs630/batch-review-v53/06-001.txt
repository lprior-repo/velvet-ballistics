12: /// Atomic cross-keyspace write batch backed by Fjall.
13: ///
14: /// Accumulates writes across multiple keyspaces and commits them
15: /// atomically with a single WAL fsync.
16: ///
17: /// # Invariant I1
18: /// `JournalWriteBatch` is `!Send + !Sync` because it contains
19: /// `PhantomData<*mut FjallJournal>` which is `!Send + !Sync`,
20: /// preventing any batch handle from crossing thread boundaries.
21: pub struct JournalWriteBatch<'j> {
22:     pub(super) inner: fjall::OwnedWriteBatch,
23:     pub(super) journal: &'j FjallJournal,
24:     #[allow(dead_code)]
25:     pub(super) staged_event_keys: HashSet<[u8; JOURNAL_KEY_BYTES]>,
26:     pub(super) aborted: bool,
27:     pub(super) staged_bytes: u64,
28:     pub(super) staged_metadata_bytes: u64,
29:     pub(super) byte_limit: Option<u64>,
30:     pub(super) _not_send_or_sync: core::marker::PhantomData<*mut FjallJournal>,
31: }
32: 
33: impl<'j> JournalWriteBatch<'j> {
34:     /// Creates a new batch for the given journal.
35:     pub fn new(journal: &'j FjallJournal) -> Self {
36:         Self {
37:             inner: journal.database.batch(),

40:             aborted: false,
41:             staged_bytes: 0,
42:             staged_metadata_bytes: 0,

48:     /// Returns the number of operations in the batch.
49:     #[must_use]
50:     pub fn len(&self) -> usize {
51:         if self.aborted { 0 } else { self.inner.len() }
52:     }
53: 
54:     /// Returns true if the batch contains no operations.
55:     #[must_use]
56:     pub fn is_empty(&self) -> bool {
57:         self.len() == 0
58:     }

60:     /// Returns true if the batch is in the aborted state.
61:     ///
62:     /// A batch becomes aborted when a fallible step sets
63:     /// `self.aborted = true` before propagating a typed error
64:     /// (`DuplicateEvent`, `KeyCapacity`, `PayloadTooLarge`, etc.).
65:     /// The `commit()` short-circuit then refuses to persist the
66:     /// partial batch. This accessor is the canonical way to
67:     /// distinguish an aborted batch from a fresh empty batch (both
68:     /// have `len() == 0`).
69:     #[must_use]
70:     pub fn is_aborted(&self) -> bool {

72:     }

79:     }
