11: /// Duplicate run submission fails with RunAlreadyExists before counter change.
12: #[test]
13: fn cli_duplicate_run_rejected_before_counter_publication() -> TestResult {
14:     let (_dir, journal) = open_test_journal()?;
15:     let submission = make_submission(RuntimePolicy::Strict, 1)?;
16:     journal.commit_run_submission(&submission, RuntimePolicy::Strict)?;
17:     assert_eq!(read_counter(&journal)?, Some(1));
18: 
19:     let before_inputs = journal.run_inputs(vb_core::RunId::new(100))?;
20:     let before_events = journal.events_for_run(vb_core::RunId::new(100))?;
21:     let before_snapshot = snapshot_journal(&journal)?;
22: 
23:     let result = journal.commit_run_submission(&submission, RuntimePolicy::Strict);
24:     assert!(
25:         matches!(result, Err(JournalError::RunAlreadyExists)),
26:         "expected RunAlreadyExists, got: {result:?}"
27:     );
28: 
29:     // Full journal state unchanged on duplicate: all typed records and index rows identical
30:     let after_snapshot = snapshot_journal(&journal)?;
31:     assert_eq!(after_snapshot, before_snapshot);
32: 
33:     // Old field-level assertions still hold
34:     assert_eq!(read_counter(&journal)?, Some(1));
35:     assert_eq!(journal.run_inputs(vb_core::RunId::new(100))?, before_inputs);
36:     assert_eq!(
37:         journal.events_for_run(vb_core::RunId::new(100))?,
38:         before_events
39:     );
40: 
41:     // Verify a fresh distinct run still commits correctly after duplicate
42:     let submission2 = make_submission_run200(RuntimePolicy::Strict, 2)?;
43:     journal.commit_run_submission(&submission2, RuntimePolicy::Strict)?;
44:     assert_eq!(read_counter(&journal)?, Some(2));
45:     verify_cli_submission_artifact_seq(&journal, vb_core::RunId::new(200), 2)?;
46:     Ok(())
47: }
