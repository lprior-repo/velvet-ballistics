450: /// Decodes a raw journal-event key byte slice into its typed
451: /// `StorageKey::RunEvent { run, seq }` form.
452: ///
453: /// Convenience wrapper around [`decode_storage_key`] that narrows the
454: /// result to the run-event variant the journal event keyspace uses,
455: /// and converts [`KeyDecodeError`] into [`JournalError`] so replay
456: /// loops do not need a second `match` arm.
457: ///
458: /// # Errors
459: ///
460: /// Returns [`JournalError::MalformedKeyspaceRow`] for any
461: /// `KeyDecodeError` (length mismatch, unknown prefix, invalid run id,
462: /// reserved seq sentinel). Returns [`JournalError::MalformedKeyspaceRow`]
463: /// with `actual_len == 17` shape when the prefix/format is fine but the
464: /// stored shape is a different variant (e.g. RunHeader) than RunEvent.
465: pub fn decode_run_event_key(
466:     bytes: &[u8],
467:     expected_run: RunId,
468: ) -> Result<(RunId, EventSeq), JournalError> {
469:     decode_storage_key(bytes)
470:         .map_err(|_err| JournalError::MalformedKeyspaceRow {
471:             prefix: PREFIX_RUN_EVENT,
472:             expected_len: JOURNAL_KEY_BYTES,
473:             actual_len: bytes.len(),
474:         })
475:         .and_then(|key| match key {
476:             StorageKey::RunEvent { run, seq } if run == expected_run => Ok((run, seq)),
477:             StorageKey::RunEvent { .. } => Err(JournalError::MalformedKeyspaceRow {
478:                 prefix: PREFIX_RUN_EVENT,
479:                 expected_len: JOURNAL_KEY_BYTES,
480:                 actual_len: bytes.len(),
481:             }),
482:             _ => Err(JournalError::MalformedKeyspaceRow {
483:                 prefix: PREFIX_RUN_EVENT,
484:                 expected_len: JOURNAL_KEY_BYTES,
485:                 actual_len: bytes.len(),
486:             }),
487:         })
488: }
