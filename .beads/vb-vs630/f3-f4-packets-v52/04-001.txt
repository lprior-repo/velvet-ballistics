66: // Case 3: Same journal, three policies increment sequence.
67: #[test]
68: fn sequential_submissions_increment() -> TestResult {
69:     let fixture = PackedFixture::new()?;
70:     let (_dir, journal) = open_test_journal()?;
71:     let strict = fixture.submit(&journal, RuntimePolicy::Strict)?;
72:     assert_eq!(strict.accepted_at_seq.get(), 1);
73:     let journaled = fixture.submit(&journal, RuntimePolicy::Journaled)?;
74:     assert_eq!(journaled.accepted_at_seq.get(), 2);
75:     let relaxed = fixture.submit(&journal, RuntimePolicy::Relaxed)?;
76:     assert_eq!(relaxed.accepted_at_seq.get(), 3);
77:     assert_ne!(strict.digest, journaled.digest);
78:     assert_ne!(journaled.digest, relaxed.digest);
79:     assert_ne!(strict.digest, relaxed.digest);
80:     fixture.check(&journal, &strict, RuntimePolicy::Strict)?;
81:     fixture.check(&journal, &journaled, RuntimePolicy::Journaled)?;
82:     fixture.check(&journal, &relaxed, RuntimePolicy::Relaxed)?;
83:     assert_eq!(read_counter(&journal)?, Some(3));
84:     Ok(())
85: }
