101: pub fn index_status_key(
102:     state: crate::types::IndexStatusState,
103:     timestamp: u64,
104:     run: RunId,
105: ) -> Result<[u8; INDEX_STATUS_KEY_BYTES], JournalError> {
106:     // PO-cn2v4-001: a zero `RunId` is reserved as "no run" per the
107:     // storage contract. Reject it before any byte writing so the typed
108:     // error surfaces exactly once at the encoder boundary (the
109:     // decoder in `decode_storage_key` mirrors this check). This is
110:     // checked BEFORE the IndexStatusState collision check so the
111:     // encoder ordering matches what the decoder reports when both
112:     // conditions hold.
113:     require_non_zero_run(run)?;
114:     // VB-NOORE (wildcard elimination): use `to_u8_checked` so an
115:     // `Other(v)` whose byte is in the collision range
116:     // `0..MIN_OTHER_STATUS_BYTE` is rejected with a typed
117:     // `JournalError::IndexStatusStateCollision` instead of silently
118:     // emitting a collision byte (SC-001 / vb-f1xkn).
119:     let state_byte = state.to_u8_checked()?;
120:     let mut key = ArrayVec::<u8, INDEX_STATUS_KEY_BYTES>::new();
121:     key.try_push(PREFIX_INDEX_STATUS)
122:         .map_err(|_| JournalError::KeyCapacity)?;
123:     key.try_push(state_byte)
124:         .map_err(|_| JournalError::KeyCapacity)?;
125:     key.try_extend_from_slice(&timestamp.to_be_bytes())
126:         .map_err(|_| JournalError::KeyCapacity)?;
127:     key.try_extend_from_slice(&run.get().to_be_bytes())
128:         .map_err(|_| JournalError::KeyCapacity)?;
129:     key.into_inner().map_err(|_| JournalError::KeyCapacity)
130: }
