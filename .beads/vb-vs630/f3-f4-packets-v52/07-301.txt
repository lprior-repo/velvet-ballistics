337: /// Decodes a raw byte slice into a typed `StorageKey`.
338: ///
339: /// 1. Classifies the prefix byte via `try_key_prefix`.
340: /// 2. Validates the key length matches the expected byte count for that prefix.
341: /// 3. Decodes numeric fields from big-endian bytes.
342: /// 4. Validates domain rules (RunId must be non-zero; EventSeq must not be MAX).
343: ///
344: /// # Errors
345: ///
346: /// Returns a `KeyDecodeError` variant for empty input, unknown prefix,
347: /// length mismatch, invalid RunId (zero), or reserved seq sentinel (MAX).
348: ///
349: /// # Examples
350: ///
351: /// ```
352: /// use vb_storage::keys::decode_storage_key;
353: /// use vb_storage::StorageKey;
354: /// use vb_core::RunId;
355: ///
356: /// let bytes = [0x10, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x2A];
357: /// let expected = StorageKey::RunHeader { run: RunId::new(42) };
358: /// assert!(decode_storage_key(&bytes).is_ok_and(|k| k == expected));
359: /// ```

365:             prefix: prefix.to_u8(),
