548: /// Validates that `run` is non-zero.
549: ///
550: /// Zero is reserved as the "no run" sentinel and must not be encoded
551: /// into any storage key. The decoder in `decode_storage_key` already
552: /// rejects this on the read path; the encoders mirror the same rule
553: /// so storage never silently accepts a zero run id at write time
554: /// and a later decode would surface a different key shape.
555: ///
556: /// PO-cn2v4-001: zero `RunId` rejection at every storage-key encoder
557: /// boundary; defence-in-depth against accidental all-zero run
558: /// encoding that would collide with no-row sentinel keys.
559: #[inline]
560: fn require_non_zero_run(run: RunId) -> Result<(), JournalError> {
561:     if run.get() == 0 {
562:         return Err(JournalError::InvalidRunId { run });
563:     }
564:     Ok(())
565: }
