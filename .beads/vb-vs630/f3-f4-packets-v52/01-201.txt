242: /// Bounded concurrent distinct-run CLI submissions produce distinct
243: /// sequences with no collision.
244: #[test]
245: fn cli_bounded_concurrent_submissions() -> TestResult {
246:     let (_dir, journal) = open_test_journal()?;
247:     let journal = Arc::new(journal);
248: 
249:     let run_configs = [
250:         (vb_core::RunId::new(100), 10),
251:         (vb_core::RunId::new(101), 11),
252:         (vb_core::RunId::new(102), 12),
253:         (vb_core::RunId::new(103), 13),
254:     ];
255: 
256:     let mut handles = Vec::new();
257:     for (run, value) in run_configs {
258:         let journal_clone = Arc::clone(&journal);
259:         let handle = std::thread::spawn(move || {
260:             let submission = make_submission_for_run(RuntimePolicy::Strict, value, run)
261:                 .map_err(|e| format!("make submission failed: {e}"))?;
262:             journal_clone
263:                 .commit_run_submission(&submission, RuntimePolicy::Strict)
264:                 .map_err(|e| format!("commit failed: {e}"))
265:         });
266:         handles.push(handle);
267:     }
268: 
269:     for handle in handles.into_iter() {
270:         handle
271:             .join()
272:             .map_err(|_| "submission thread panicked")?
273:             .map_err(|e| format!("concurrent submission failed: {e}"))?;
274:     }
275: 
276:     assert_eq!(read_counter(&journal)?, Some(4));
277: 
278:     // Verify each run got a distinct sequence
279:     let mut seqs = Vec::new();
280:     for (run, _) in run_configs {
281:         let inputs = journal.run_inputs(run)?.ok_or("inputs not found")?;
282:         let compiled = journal
283:             .compiled_ir(inputs.artifact_digest())?
284:             .ok_or("compiled not found")?;
285:         let decoded: crate::AcceptedArtifact = postcard::from_bytes(&compiled.ir)?;
286:         seqs.push(decoded.accepted_at_seq.get());
287:     }
288: 
289:     seqs.sort();
290:     assert_eq!(seqs, [1, 2, 3, 4]);
291:     Ok(())
292: }
