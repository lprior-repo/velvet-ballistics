23: /// Policy that decides how a keyspace iteration handles rows whose key
24: /// cannot be parsed back into a typed `StorageKey` (truncated bytes,
25: /// unknown prefix, or shape mismatch).
26: ///
27: /// Picking a single policy per call site is the unifying rule for CC-002:
28: /// production paths default to [`Self::FailClosed`]; diagnostic/doctor
29: /// callers that must keep scanning across partial corruption opt in to
30: /// [`Self::SkipMalformed`]. Never use `SkipMalformed` on a production
31: /// read path that is supposed to surface evidence of corruption.
32: #[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
33: pub enum KeyspaceScanPolicy {
34:     /// Surface the first malformed row as a typed
35:     /// [`crate::JournalError::MalformedKeyspaceRow`] error and abort the
36:     /// scan. This is the default for production paths.
37:     #[default]
38:     FailClosed,
39:     /// Skip malformed rows silently and continue scanning. Reserved for
40:     /// doctor/diagnostic tooling that must produce a partial view of a
41:     /// partially-corrupt keyspace. The caller is responsible for
42:     /// reporting the skipped count to the operator; the policy itself
43:     /// does not log.
44:     SkipMalformed,
45: }
46: 
47: impl KeyspaceScanPolicy {
48:     /// Returns the policy used by production scan APIs by default.
49:     pub const fn default_production() -> Self {
50:         Self::FailClosed
51:     }
52: 
53:     /// Returns the policy used by doctor/diagnostic scan APIs by
54:     /// default. Doctor paths must tolerate partial corruption so the
55:     /// operator can see whatever is still well-formed.
56:     pub const fn default_doctor() -> Self {
57:         Self::SkipMalformed
58:     }
59: }
