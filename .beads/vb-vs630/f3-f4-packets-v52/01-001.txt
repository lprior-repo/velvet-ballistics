26: fn check_malformed_counter(bytes: &[u8], expected_actual_len: usize) -> TestResult {
27:     let (_dir, journal) = open_test_journal()?;
28:     write_malformed_counter(&journal, bytes)?;
29:     // Verify the raw malformed counter is stored as-is (don't call production read_counter,
30:     // which returns None on length mismatch and would exit early)
31:     let before = journal
32:         .artifact_acceptance
33:         .get(b"last_issued")?
34:         .ok_or("counter row missing")?;
35:     assert_eq!(before.as_ref(), bytes);
36: 
37:     let submission = make_submission(RuntimePolicy::Strict, 99)?;
38:     let result = journal.commit_run_submission(&submission, RuntimePolicy::Strict);
39:     match result {
40:         Err(JournalError::MalformedKeyspaceRow {
41:             prefix,
42:             expected_len,
43:             actual_len,
44:         }) => {
45:             assert_eq!(prefix, 0);
46:             assert_eq!(expected_len, 8);
47:             assert_eq!(actual_len, expected_actual_len);
48:         }
49:         other => return Err(format!("expected MalformedKeyspaceRow, got: {other:?}").into()),
50:     }
51:     // Verify counter unchanged and journal otherwise empty
52:     let after = journal
53:         .artifact_acceptance
54:         .get(b"last_issued")?
55:         .ok_or("counter row missing")?;
56:     assert_eq!(after.as_ref(), before.as_ref());
57:     verify_empty_journal(&journal)?;
58:     Ok(())
59: }
