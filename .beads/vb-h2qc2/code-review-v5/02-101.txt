125: pub fn fuzz_span_bridge(data: &[u8]) {
126:     use vb_compile::{SourceSpan, build_semantic_source_map, build_source_map};
127: 
128:     let mut values = [0usize; 6];
129:     for (slot, byte) in values.iter_mut().zip(data.iter().copied()) {
130:         *slot = usize::from(byte);
131:     }
132: 
133:     let span = SourceSpan::new(
134:         values[0], values[1], values[2], values[3], values[4], values[5],
135:     );
136:     assert_eq!(span.start_offset, values[0]);
137:     assert_eq!(span.end_offset, values[1]);
138:     assert_eq!(span.start_line, values[2]);
139:     assert_eq!(span.start_col, values[3]);
140:     assert_eq!(span.end_line, values[4]);
141:     assert_eq!(span.end_col, values[5]);
142: 
143:     let Ok(text) = std::str::from_utf8(data) else {
144:         return;
145:     };
146: 
147:     if let Ok(map) = build_source_map(text) {
148:         for (index, mapped_span) in map.iter() {
149:             assert_eq!(map.span_for_node(index), Some(mapped_span));
150:         }
151:     }
152: 
153:     if let Ok(map) = build_semantic_source_map(text) {
154:         let _root = map.span_for_path("$");
155:         let _manual = map.span_for_path("$.when.manual");
156:         let _step_zero = map.span_for_path("$.steps[0]");
157:         // `_root`, `_manual`, and `_step_zero` are deliberately bound to
158:         // underscore-prefixed identifiers so the strict
159:         // `let_underscore_must_use` lint stays triggered only on
160:         // bare `let _ = must_use_call()`. They prove the lookup table
161:         // returns a span option without exercising it.
162:     }
163: }
