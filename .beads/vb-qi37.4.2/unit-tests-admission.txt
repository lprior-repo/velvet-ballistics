// =======================================================================
// Unit Tests: Admission Rejection (UNIT-ADMIT-001, UNIT-ADMIT-002)
// =======================================================================
// Location: crates/vb_runtime/src/admission.rs (add to #[cfg(test)] module near line 890)
// Prerequisites: NeverPresentArtifactStore must be implemented in admission.rs

#[cfg(test)]
mod admission_rejection_tests {
    use super::*;

    /// Test artifact store that always returns ArtifactNotFound.
    /// Mirrors the NeverPresentArtifactStore production type but scoped to tests.
    struct NeverPresentTestStore;
    impl AcceptedArtifactStore for NeverPresentTestStore {
        fn load_accepted_artifact(
            &self,
            digest: WorkflowDigest,
        ) -> Result<vb_storage::admission::AcceptedArtifact, ArtifactEnvelopeError> {
            Err(ArtifactEnvelopeError::ArtifactNotFound { digest })
        }
    }

    #[test]
    fn admit_run_strict_without_artifact_rejected() {
        // UNIT-ADMIT-001: Strict policy rejects missing artifact
        let store = NeverPresentTestStore;
        let digest = WorkflowDigest::from_bytes([42u8; 32]);
        let run_id = RunId::new(1);
        let caps = CapabilitySet::empty();
        let policy = RuntimePolicy::Strict;

        let result = admit_artifact_run(&store, policy, run_id, digest, caps);

        assert!(
            matches!(result, Err(AdmissionError::ArtifactNotFound { .. })),
            "Strict policy must reject missing artifact with ArtifactNotFound, got {:?}",
            result
        );
    }

    #[test]
    fn admit_run_journaled_without_artifact_rejected() {
        // UNIT-ADMIT-002: Journaled policy rejects missing artifact
        let store = NeverPresentTestStore;
        let digest = WorkflowDigest::from_bytes([42u8; 32]);
        let run_id = RunId::new(2);
        let caps = CapabilitySet::empty();
        let policy = RuntimePolicy::Journaled;

        let result = admit_artifact_run(&store, policy, run_id, digest, caps);

        assert!(
            matches!(result, Err(AdmissionError::ArtifactNotFound { .. })),
            "Journaled policy must reject missing artifact with ArtifactNotFound, got {:?}",
            result
        );
    }

    #[test]
    fn admit_run_relaxed_skips_artifact_validation() {
        // Relaxed policy bypasses artifact existence check
        let store = NeverPresentTestStore;
        let digest = WorkflowDigest::from_bytes([42u8; 32]);
        let run_id = RunId::new(3);
        let caps = CapabilitySet::empty();
        let policy = RuntimePolicy::Relaxed;

        let result = admit_artifact_run(&store, policy, run_id, digest, caps);

        assert!(
            result.is_ok(),
            "Relaxed policy must accept even with missing artifact, got {:?}",
            result
        );
    }

    #[test]
    fn admit_run_strict_requires_all_proof_flags() {
        // Strict policy validates proof flags; AlwaysPresentArtifactStore returns
        // a fully-valid artifact in tests. This test verifies the positive path.
        let store = AlwaysPresentArtifactStore;
        let digest = WorkflowDigest::from_bytes([99u8; 32]);
        let run_id = RunId::new(4);
        let caps = CapabilitySet::empty();
        let policy = RuntimePolicy::Strict;

        let result = admit_artifact_run(&store, policy, run_id, digest, caps);

        assert!(
            result.is_ok(),
            "Strict policy must accept valid artifact with all proof flags, got {:?}",
            result
        );
    }
}
