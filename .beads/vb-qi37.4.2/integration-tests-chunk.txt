#[test]
fn admission_rejection_does_not_insert_run_state_strict() -> Result<(), String> {
    // INT-INV-001: Strict policy + NeverPresentArtifactStore → run NOT inserted
    // Verifies POST-002: on rejection, no frame allocated, no run state inserted, counter unchanged
    use crate::admission::{AcceptedArtifactStore, NeverPresentArtifactStore};
    use crate::journal::NoopRuntimeJournal;
    use vb_core::ids::RunId;
    use vb_core::policy::RuntimePolicy;

    let config = ShardConfig {
        command_queue_capacity: 16,
        trace_capacity: 16,
        step_budget_per_tick: 4,
        max_active_runs: 4,
        policy: RuntimePolicy::Strict,
    };
    let shard = Shard::new_with_journal_and_artifact_store(
        config,
        NoopRuntimeJournal::shared(),
        NeverPresentArtifactStore::shared(),
    );
    let mut shard = shard;
    let workflow = require_workflow("suspended", suspended_workflow())?;
    let run = RunId::new(100);

    // Enqueue submit command
    assert_eq!(
        shard.enqueue(ShardCommand::Submit {
            run,
            workflow,
            caps: CapabilitySet::empty(),
        }),
        Ok(())
    );

    // tick() processes the command — build_admission returns Err(ArtifactNotFound)
    // which propagates via ? and prevents all subsequent steps
    let result = shard.tick();

    // Assert correct error type
    assert!(
        matches!(result, Err(RuntimeError::AdmissionArtifactNotFound { .. })),
        "expected AdmissionArtifactNotFound, got {:?}",
        result
    );

    // INV-001: run was NOT inserted into self.runs
    assert_eq!(
        shard.active_run_count(),
        0,
        "run must not be inserted on rejection (INV-001)"
    );

    // POST-002: runs_submitted counter must NOT increment on rejection
    assert_eq!(
        shard.counters().snapshot().runs_submitted,
        0,
        "runs_submitted counter must remain 0 on rejection (POST-002)"
    );

    Ok(())
}

#[test]
fn admission_capability_mismatch_does_not_insert() -> Result<(), String> {
    // INT-ERR-001: Strict policy + AlwaysPresentArtifactStore + insufficient caps
    // → AdmissionCapabilityDenied, run NOT inserted
    use crate::admission::AlwaysPresentArtifactStore;
    use crate::journal::NoopRuntimeJournal;
    use vb_core::ids::RunId;
    use vb_core::policy::RuntimePolicy;

    let config = ShardConfig {
        command_queue_capacity: 16,
        trace_capacity: 16,
        step_budget_per_tick: 4,
        max_active_runs: 4,
        policy: RuntimePolicy::Strict,
    };
    // AlwaysPresentArtifactStore: artifact exists but we submit with empty CapabilitySet
    let shard = Shard::new_with_journal_and_artifact_store(
        config,
        NoopRuntimeJournal::shared(),
        AlwaysPresentArtifactStore::shared(),
    );
    let mut shard = shard;
    let workflow = require_workflow("suspended", suspended_workflow())?;
    let run = RunId::new(102);

    // suspended_workflow has action: ActionId::new(0) which requires a capability
    // We submit with EMPTY caps → CapabilityDenied
    assert_eq!(
        shard.enqueue(ShardCommand::Submit {
            run,
            workflow,
            caps: CapabilitySet::empty(), // No capabilities granted
        }),
        Ok(())
    );

    let result = shard.tick();
    assert!(
        matches!(result, Err(RuntimeError::AdmissionCapabilityDenied { .. })),
        "expected AdmissionCapabilityDenied, got {:?}",
        result
    );

    assert_eq!(
        shard.active_run_count(),
        0,
        "run must not be inserted on capability denial (INV-001)"
    );

    Ok(())
}

#[test]
fn admission_rejection_no_counter_increment() -> Result<(), String> {
    // INT-POST-001: Rejection must NOT increment runs_submitted counter
    use crate::admission::{AcceptedArtifactStore, NeverPresentArtifactStore};
    use crate::journal::NoopRuntimeJournal;
    use vb_core::ids::RunId;
    use vb_core::policy::RuntimePolicy;

    let config = ShardConfig {
        command_queue_capacity: 16,
        trace_capacity: 16,
        step_budget_per_tick: 4,
        max_active_runs: 4,
        policy: RuntimePolicy::Strict,
    };
    let shard = Shard::new_with_journal_and_artifact_store(
        config,
        NoopRuntimeJournal::shared(),
        NeverPresentArtifactStore::shared(),
    );
    let mut shard = shard;
    let workflow = require_workflow("suspended", suspended_workflow())?;
    let run = RunId::new(103);

    // Record initial counter state
    let initial_submitted = shard.counters().snapshot().runs_submitted;

    assert_eq!(
        shard.enqueue(ShardCommand::Submit {
            run,
            workflow,
            caps: CapabilitySet::empty(),
        }),
        Ok(())
    );

    let result = shard.tick();
    assert!(
        matches!(result, Err(RuntimeError::AdmissionArtifactNotFound { .. })),
        "expected AdmissionArtifactNotFound, got {:?}",
        result
    );

    // POST-002: counter must NOT have incremented
    let final_submitted = shard.counters().snapshot().runs_submitted;
    assert_eq!(
        final_submitted, initial_submitted,
        "runs_submitted counter must not change on rejection (POST-002)"
    );
    assert_eq!(
        final_submitted, 0,
        "runs_submitted must remain 0 after rejected submit"
    );

    Ok(())
}
