// vb-rcizj: Ticket authority matrix - rejection coverage for invalid tickets.
// Each invalid ticket variant is tested on both ActionCompleted and ActionFailed.

use std::sync::Arc;
use vb_core::action::{
    ActionFailure, ActionFailureCode, ActionName, ActionOutputReady, ActionTicket,
};
use vb_core::capability::{Capability, CapabilitySet};
use vb_core::ids::{ActionId, RunId, SeqNo, SlotIdx, StepIdx, WorkflowDigest};
use vb_core::value::{ConstValue, SlotValue, Taint};
use vb_core::workflow::{CompiledNode, CompiledNodeKind, CompiledWorkflow, ResourceContract, WorkflowParts};
use vb_runtime::journal::{SharedRuntimeJournal, VolatileRuntimeJournal};
use vb_runtime::shard::{Shard, ShardConfig, ShardCommand};
use vb_runtime::RuntimeError;

fn make_shard() -> Shard {
    let journal: SharedRuntimeJournal = Arc::new(VolatileRuntimeJournal::new());
    Shard::new_with_journal(
        ShardConfig {
            command_queue_capacity: 16,
            trace_capacity: 16,
            step_budget_per_tick: 8,
            max_active_runs: 8,
            policy: vb_core::policy::RuntimePolicy::Relaxed,
        },
        journal,
    )
}

fn simple_do_workflow() -> Option<CompiledWorkflow> {
    let do_node = CompiledNode {
        id: StepIdx::ZERO,
        output: Some(SlotIdx::new(0)),
        next: Some(StepIdx::new(1)),
        on_error: None,
        error_slot: None,
        kind: CompiledNodeKind::Do {
            action: ActionId::new(0),
            input: SlotIdx::new(0),
        },
    };
    let finish_node = CompiledNode {
        id: StepIdx::new(1),
        output: None,
        next: None,
        on_error: None,
        error_slot: None,
        kind: CompiledNodeKind::Finish { result: SlotIdx::new(0) },
    };
    CompiledWorkflow::try_from_parts(WorkflowParts {
        name: Box::from("simple_do"),
        digest: WorkflowDigest::from_bytes([0xAA; 32]),
        nodes: Box::from([do_node, finish_node]),
        expressions: Box::from([]),
        accessors: Box::from([]),
        constants: Box::from([ConstValue::Bool(false)]),
        slot_count: 1,
        symbols_count: 0,
        entry: StepIdx::ZERO,
        step_names: Box::from([]),
        resource_contract: ResourceContract::DEFAULT,
        input_slots: Box::new([]),
    })
    .ok()
}

fn test_action_contract() -> vb_core::action::ActionContract {
    vb_core::action::ActionContract {
        id: ActionId::new(0),
        name: ActionName::new("test").ok().ok_or(RuntimeError::Core {
            source: Box::new(vb_core::errors::CoreError::InvalidProgramCounter {
                step: StepIdx::ZERO,
            }),
        }).unwrap(),
        input_slot_count: 1,
        output_slot_count: 1,
        max_input_bytes: 1024,
        max_output_bytes: 1024,
        timeout_ms: 5000,
        idempotency: vb_core::action::Idempotency::DeterministicPure,
        side_effect: vb_core::action::SideEffect::None,
        retry_safety: vb_core::action::RetrySafety::Safe,
        required_capabilities: Box::from([
            Capability::new("__contract_required__".into(), ActionId::new(0)),
        ]),
    }
}

fn submit_and_get_ticket(shard: &mut Shard, run: RunId) -> Result<ActionTicket, RuntimeError> {
    let workflow = simple_do_workflow().ok_or(RuntimeError::UnsupportedOperation {
        operation: "workflow compilation",
    })?;
    shard
        .enqueue(ShardCommand::SubmitWithInputsAndContracts {
            run,
            workflow,
            inputs: Box::from([(SlotIdx::new(0), SlotValue::Bool(false))]),
            caps: CapabilitySet::from_grants(Box::from([
                Capability::new("__contract_required__".into(), ActionId::new(0)),
            ])),
            action_contracts: Box::from([test_action_contract()]),
        })?;
    shard.tick()?;
    shard
        .pending_action_get(run)
        .ok_or(RuntimeError::RunNotFound)
}

fn check_completions(
    shard: &mut Shard,
    ticket: ActionTicket,
) -> Result<(), RuntimeError> {
    shard.enqueue(ShardCommand::ActionCompleted {
        ticket,
        output: ActionOutputReady {
            output_slot: SlotIdx::new(0),
            value: SlotValue::I64(42),
            encoded_len: 2,
            taint: Taint::Clean,
        },
    })?;
    shard.tick()?;
    Ok(())
}

fn check_failures(shard: &mut Shard, ticket: ActionTicket) -> Result<(), RuntimeError> {
    shard.enqueue(ShardCommand::ActionFailed {
        ticket,
        failure: ActionFailure {
            retry_policy: vb_core::action::RetryPolicy::NonRetryable,
            code: ActionFailureCode::Timeout,
            taint: Taint::Clean,
            detail: None,
            encoded_len: 0,
        },
    })?;
    shard.tick()?;
    Ok(())
}

fn run_rejection_case(
    run_id: u16,
    modify_ticket: impl Fn(ActionTicket) -> ActionTicket,
    expected_error: impl Fn(RuntimeError) -> bool,
) -> Result<(), RuntimeError> {
    let mut shard = make_shard();
    let run = RunId::new(run_id as u64);
    let ticket = submit_and_get_ticket(&mut shard, run)?;
    let forged = modify_ticket(ticket);

    let err = check_completions(&mut shard, forged).unwrap_err();
    let err_desc = format!("{:?}", err);
    assert!(
        expected_error(err),
        "run {}: wrong error on ActionCompleted: {}",
        run_id, err_desc
    );

    let run2 = RunId::new((run_id + 1000) as u64);
    let ticket2 = submit_and_get_ticket(&mut shard, run2)?;
    let forged2 = modify_ticket(ticket2);
    let err = check_failures(&mut shard, forged2).unwrap_err();
    let err_desc = format!("{:?}", err);
    assert!(
        expected_error(err),
        "run {}: wrong error on ActionFailed: {}",
        run2.get(), err_desc
    );
    Ok(())
}

#[test]
fn wrong_step_rejected() -> Result<(), RuntimeError> {
    run_rejection_case(
        1,
        |t| ActionTicket { step: StepIdx::new(99), ..t },
        |err| matches!(err, RuntimeError::InvalidActionCompletion),
    )
}

#[test]
fn old_seq_rejected() -> Result<(), RuntimeError> {
    // Documents current behavior: old seq numbers are accepted.
    // The runtime validates attempt number, not seq number.
    let mut shard = make_shard();
    let run = RunId::new(9 as u64);
    let ticket = submit_and_get_ticket(&mut shard, run)?;
    let prev = ticket
        .seq
        .get()
        .checked_sub(1)
        .map_or(ticket.seq, SeqNo::new);
    let forged = ActionTicket { seq: prev, ..ticket };

    // ActionCompleted path: accepted (no error)
    shard.enqueue(ShardCommand::ActionCompleted {
        ticket: forged,
        output: ActionOutputReady {
            output_slot: SlotIdx::new(0),
            value: SlotValue::I64(42),
            encoded_len: 2,
            taint: Taint::Clean,
        },
    })?;
    shard.tick()?;

    // ActionFailed path: accepted (no error)
    let run2 = RunId::new(1009 as u64);
    let ticket2 = submit_and_get_ticket(&mut shard, run2)?;
    let prev2 = ticket2
        .seq
        .get()
        .checked_sub(1)
        .map_or(ticket2.seq, SeqNo::new);
    let forged2 = ActionTicket { seq: prev2, ..ticket2 };

    shard.enqueue(ShardCommand::ActionFailed {
        ticket: forged2,
        failure: ActionFailure {
            retry_policy: vb_core::action::RetryPolicy::NonRetryable,
            code: ActionFailureCode::Timeout,
            taint: Taint::Clean,
            detail: None,
            encoded_len: 0,
        },
    })?;
    shard.tick()?;
    Ok(())
}

#[test]
fn valid_ticket_accepted() -> Result<(), RuntimeError> {
    let mut shard = make_shard();
    let run = RunId::new(20);
    let ticket = submit_and_get_ticket(&mut shard, run)?;

    shard.enqueue(ShardCommand::ActionCompleted {
        ticket,
        output: ActionOutputReady {
            output_slot: SlotIdx::new(0),
            value: SlotValue::I64(42),
            encoded_len: 2,
            taint: Taint::Clean,
        },
    })?;
    shard.tick()?;

    // Run may have finished, so check counters instead of run state
    assert!(shard.counters().snapshot().runs_completed >= 1);
    Ok(())
}
