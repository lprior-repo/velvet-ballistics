232:     step: &Yaml<'_>,
233:     index: usize,

258:             builder,

260:         StepPrimitive::Choose => {
261:             compile_choose(body, index, last_step, id, source_ir_starts, builder)
262:         }
263:         StepPrimitive::ForEach => return compile_for_each(body, index, last_step, id, builder),
264:         StepPrimitive::Parallel => {

278:         StepPrimitive::Finish => return compile_finish(body, index, last_step, id, builder),
