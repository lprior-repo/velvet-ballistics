175: fn verify_failure_path(result: Result<Vec<CompiledNode>, CompileError>, builder: &SlotCompiler) {
176:     match result {
177:         Ok(nodes) => {
178:             std::mem::forget(nodes);
179:             kani::assert(false, "lower_repeat should fail for prefix > 65533");
180:         }
181:         Err(CompileError::SlotIndexOutOfRange { value }) => {
182:             kani::assert(value == 65535, "overflow error must report value 65535");
183:         }
184:         Err(_) => kani::assert(false, "expected SlotIndexOutOfRange error"),
185:     }
186:     match builder.slot_count() {
187:         Ok(actual_count) => {
188:             kani::assert(actual_count == 65535, "slot count must be 65535 on failure")
189:         }
190:         Err(_) => kani::assert(false, "slot_count must succeed after failed lower_repeat"),
191:     }
192: }

194: fn cover_repeat_inputs(prefix: u16, max_attempts: u16, body: StepIdx, done: StepIdx, id: StepIdx) {
195:     kani::cover!(prefix == 0, "prefix domain includes zero");
196:     kani::cover!(
197:         prefix == 65533,
198:         "prefix domain includes max success boundary"
199:     );
200:     kani::cover!(
201:         prefix == 65534,
202:         "prefix domain includes first failure value"
203:     );
204:     kani::cover!(prefix == 65535, "prefix domain includes u16 max");
205:     kani::cover!(max_attempts == 0, "max_attempts includes zero");
206:     kani::cover!(max_attempts == u16::MAX, "max_attempts includes u16 max");
207:     kani::cover!(id == StepIdx::new(u16::MAX), "id domain includes u16 max");
208:     kani::cover!(
209:         body == StepIdx::new(u16::MAX),
210:         "body domain includes u16 max"
211:     );
212:     kani::cover!(
213:         done == StepIdx::new(u16::MAX),
214:         "done domain includes u16 max"
215:     );
216:     kani::cover!(body == id, "body may alias id");
217:     kani::cover!(done == body, "done may alias body");
218:     kani::cover!(done == id, "done may alias id");
