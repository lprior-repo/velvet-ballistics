#!/usr/bin/env python3
"""
Bounded Rust test fixture migration.

Converts ShardCommand::Submit / SubmitWithContracts / SubmitWithInputs
to SubmitWithInputsAndContracts for named positive test functions.

Rules:
- Preserve existing nonempty inputs exactly
- If missing/empty and Do reads slot0, provide I64(0) input
- Use explicit fixture contracts and grants for ActionId::new(0)
- Do NOT change production code, assertions, policies, or unrelated tests
"""

import pathlib
import re

ROOT = pathlib.Path("/home/lewis/src/isoloated/velvet-ballistics-vb-eyij7-integration-3090")
TESTS_DIR = ROOT / "crates/vb_runtime/src/shard/lifecycle_tests"

# Regex patterns for submit commands
SUBMIT_PATTERN = re.compile(r'ShardCommand::Submit\s*{([^}]*)}', re.DOTALL)
SUBMIT_WITH_CONTRACTS_PATTERN = re.compile(r'ShardCommand::SubmitWithContracts\s*{([^}]*)}', re.DOTALL)
SUBMIT_WITH_INPUTS_PATTERN = re.compile(r'ShardCommand::SubmitWithInputs\s*{([^}]*)}', re.DOTALL)

def extract_field(body, field_name):
    """Extract a field value from the submit command body."""
    # Try positional first (e.g., "run,")
    if field_name == 'run':
        match = re.search(r'^\s*run,\s*', body, re.MULTILINE)
        if match:
            return 'run'
    
    # Try named field (e.g., "run: RunId::new(1),")
    match = re.search(rf'{field_name}:\s*([^,\n}}]+)', body)
    if match:
        return match.group(1).strip()
    
    return None

def fix_submit(content):
    """Convert ShardCommand::Submit to SubmitWithInputsAndContracts."""
    def replacer(match):
        body = match.group(1)
        run = extract_field(body, 'run')
        workflow = extract_field(body, 'workflow')
        
        if not run or not workflow:
            return match.group(0)
        
        return f'''ShardCommand::SubmitWithInputsAndContracts {{
            {run},
            workflow: {workflow},
            inputs: Box::from([(SlotIdx::ZERO, SlotValue::I64(0))]),
            caps: crate::shard::tests::test_fixtures::contract_capability_for(ActionId::new(0)),
            action_contracts: crate::shard::tests::test_fixtures::contracts_through(ActionId::new(0)),
        }}'''
    
    return SUBMIT_PATTERN.sub(replacer, content)

def fix_submit_with_contracts(content):
    """Convert ShardCommand::SubmitWithContracts to SubmitWithInputsAndContracts."""
    def replacer(match):
        body = match.group(1)
        run = extract_field(body, 'run')
        workflow = extract_field(body, 'workflow')
        
        if not run or not workflow:
            return match.group(0)
        
        return f'''ShardCommand::SubmitWithInputsAndContracts {{
            {run},
            workflow: {workflow},
            inputs: Box::from([(SlotIdx::ZERO, SlotValue::I64(0))]),
            caps: crate::shard::tests::test_fixtures::contract_capability_for(ActionId::new(0)),
            action_contracts: crate::shard::tests::test_fixtures::contracts_through(ActionId::new(0)),
        }}'''
    
    return SUBMIT_WITH_CONTRACTS_PATTERN.sub(replacer, content)

def fix_submit_with_inputs(content):
    """Convert ShardCommand::SubmitWithInputs to SubmitWithInputsAndContracts."""
    def replacer(match):
        body = match.group(1)
        run = extract_field(body, 'run')
        workflow = extract_field(body, 'workflow')
        inputs = extract_field(body, 'inputs')
        
        if not run or not workflow:
            return match.group(0)
        
        if not inputs:
            inputs = 'Box::from([(SlotIdx::ZERO, SlotValue::I64(0))])'
        
        return f'''ShardCommand::SubmitWithInputsAndContracts {{
            {run},
            workflow: {workflow},
            inputs: {inputs},
            caps: crate::shard::tests::test_fixtures::contract_capability_for(ActionId::new(0)),
            action_contracts: crate::shard::tests::test_fixtures::contracts_through(ActionId::new(0)),
        }}'''
    
    return SUBMIT_WITH_INPUTS_PATTERN.sub(replacer, content)

def fix_file(path):
    """Apply all submit command fixes to a file."""
    content = path.read_text()
    original = content
    
    content = fix_submit(content)
    content = fix_submit_with_contracts(content)
    content = fix_submit_with_inputs(content)
    
    if content != original:
        path.write_text(content)
        print(f"Fixed {path}")
    else:
        print(f"No changes needed for {path}")

# Fix all test files
for chunk in ["chunk_003.rs", "chunk_004.rs", "chunk_005.rs", "chunk_006.rs", "chunk_009.rs"]:
    fix_file(TESTS_DIR / chunk)
