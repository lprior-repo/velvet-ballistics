/// Opens a temporary journal in a unique directory.
/// Caller must close the journal and remove the directory when done.
pub fn open_test_journal() -> Result<(std::path::PathBuf, FjallJournal), Box<dyn std::error::Error>> {
    let dir = tempfile::tempdir()?;
    let path = dir.into_path();
    let journal = FjallJournal::open(&path, None)?;
    Ok((path, journal))
}

/// Builds a minimal compiled workflow with no actions for admission testing.
/// The digest is computed correctly from the parts.
pub fn minimal_workflow() -> Result<CompiledWorkflow, Box<dyn std::error::Error>> {
    let mut parts = WorkflowParts {
        semantic_source_identity: None,
        name: Box::<str>::from("test_minimal"),
        input_slots: Box::new([]),
        digest: WorkflowDigest::from_bytes([0u8; 32]),
        nodes: Box::new([
            CompiledNode {
                id: StepIdx::new(0),
                output: Some(SlotIdx::new(0)),
                next: Some(StepIdx::new(1)),
                on_error: None,
                error_slot: None,
                kind: CompiledNodeKind::SetConst {
                    value: ConstIdx::new(0),
                },
            },
            CompiledNode {
                id: StepIdx::new(1),
                output: None,
                next: None,
                on_error: None,
                error_slot: None,
                kind: CompiledNodeKind::Finish {
                    result: SlotIdx::new(0),
                },
            },
        ]),
        expressions: Box::new([]),
        accessors: Box::new([]),
        constants: Box::new([ConstValue::I64(42)]),
        slot_count: 1,
        symbols_count: 0,
        entry: StepIdx::new(0),
        resource_contract: ResourceContract::DEFAULT,
        step_names: Box::new([]),
    };
    let hash_bytes = postcard::to_allocvec(&parts)?;
    let computed = blake3::hash(&hash_bytes);
    parts.digest = WorkflowDigest::from_bytes(computed.into());
    CompiledWorkflow::try_from_parts(parts).map_err(|e| Box::new(e) as Box<dyn std::error::Error>)
}

/// Reads the raw artifact acceptance counter from the journal.
/// Returns None if the counter key is absent.
pub fn read_counter(journal: &FjallJournal) -> Result<Option<u64>, Box<dyn std::error::Error>> {
    let raw = journal.artifact_acceptance.get(b"last_issued")?;
    match raw {
        None => Ok(None),
        Some(slice) => {
            let bytes = slice.as_ref();
            if bytes.len() != 8 {
                return Err(format!("counter has wrong length: {}", bytes.len()).into());
            }
            let arr: [u8; 8] = bytes
                .try_into()
                .map_err(|_| "counter slice conversion failed".to_string())?;
            Ok(Some(u64::from_be_bytes(arr)))
        }
    }
}

/// Writes a raw counter value to the journal for fault injection.
pub fn write_counter(journal: &FjallJournal, value: u64) -> Result<(), Box<dyn std::error::Error>> {
    journal
        .artifact_acceptance
        .insert(b"last_issued", &value.to_be_bytes())?;
    Ok(())
}

/// Verifies that a persisted artifact can be read back and matches expectations.
/// Checks: record exists, digest matches, decode matches, accepted_artifact_digest matches,
/// verification.digest matches, policy verification passes.
pub fn verify_artifact_readback(
    journal: &FjallJournal,
    digest: vb_core::WorkflowDigest,
    expected: &AcceptedArtifact,
    policy: vb_core::RuntimePolicy,
) -> Result<(), Box<dyn std::error::Error>> {
    let record = journal
        .compiled_ir(digest)?
        .ok_or("artifact record not found")?;
    if record.digest != digest {
        return Err("record digest does not match expected".into());
    }
    let decoded: AcceptedArtifact = postcard::from_bytes(&record.ir)?;
    if admission::accepted_artifact_digest(&decoded)? != decoded.digest {
        return Err("decoded artifact digest is inconsistent".into());
    }
    if decoded.verification.digest != decoded.digest {
        return Err("verification.digest mismatch".into());
    }
    if decoded != *expected {
        return Err("decoded artifact does not match expected".into());
    }
    admission::verify_accepted_artifact_for_policy(&decoded, policy)?;
    Ok(())
}
