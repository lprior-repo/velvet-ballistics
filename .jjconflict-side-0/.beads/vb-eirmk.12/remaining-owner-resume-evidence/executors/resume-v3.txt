def bytes_sha(bytes)
  Digest::SHA256.hexdigest(bytes)
end

def read_hashed(path, expected)
  bytes = File.binread(path)
  abort("SHA-256 mismatch: #{path}") unless bytes_sha(bytes) == expected
  bytes
end

def persist_bytes(path, bytes)
  File.open(path, 'wb', 0o600) do |file|
    file.write(bytes)
    file.flush
    file.fsync
  end
end

def persist_json(path, value)
  persist_bytes(path, JSON.pretty_generate(value) + "\n")
end

$commands = []
$command_number = 0
def bd(*args)
  abort('BD argument is outside the resume read/write allowlist') unless permitted_bd_args?(args)
  $command_number += 1
  stdout, stderr, status = Open3.capture3(BD_ENV, 'bd', *args, '--json', chdir: ROOT)
  receipt = { 'args' => args, 'stdout' => stdout, 'stderr' => stderr, 'exit' => status.exitstatus }
  if File.directory?(File.join(ROOT, OUT))
    persist_json(File.join(ROOT, OUT, format('bd-command-%04d.json', $command_number)), receipt)
  else
    $commands << receipt
  end
  abort("bd #{args.first} failed; stop and inspect receipts, do not retry") unless status.success?
  JSON.parse(stdout)
end

def one(value)
  value.is_a?(Array) ? value.fetch(0) : value
end

def review_receipt(path, binding_field)
  bytes = File.binread(path)
  data = JSON.parse(bytes)
  abort("review is not PASS: #{path}") unless data.fetch('verdict') == 'PASS'
  abort("review does not bind exact resume contract: #{path}") unless data.fetch(binding_field) == CONTRACT_SHA
  abort("review does not bind exact executor bytes: #{path}") unless data.fetch('executor_sha256') == EXECUTOR_SHA
  { 'path' => path, 'sha256' => bytes_sha(bytes), 'verdict' => data.fetch('verdict'),
    'binding_field' => binding_field, 'bound_sha256' => data.fetch(binding_field),
    'executor_sha256' => data.fetch('executor_sha256') }
end

def approved_payload_fields_match?(approved, observed)
  approved.all? { |field, value| observed[field] == value }
end

def exact_approved_payload?(approved, observed)
  approved.keys.sort == observed.keys.sort && approved_payload_fields_match?(approved, observed)
end

def approved_create_response_matches?(approved, response)
  returned_fields = approved.reject { |field, _| field == 'parent' }
  response_fields = response.select { |field, _| approved.key?(field) }
  response_fields == returned_fields && !response.key?('parent')
end

def assert_manifest_invariants(manifest)

end

def scalar_spans(scanner, path, targets, spans)
  scanner.skip(/\s*/)
  case scanner.peek(1)
  when '{'
    scanner.getch
    scanner.skip(/\s*/)
    return scanner.getch if scanner.peek(1) == '}'
    loop do
      key_token = scanner.scan(/"(?:[^"\\]|\\.)*"/m)
      abort('invalid design object key') unless key_token
      key = JSON.parse(key_token)
      scanner.skip(/\s*/)
      abort('missing design object colon') unless scanner.getch == ':'
      scalar_spans(scanner, path + [key], targets, spans)
      scanner.skip(/\s*/)
      separator = scanner.getch
      break if separator == '}'
      abort('missing design object comma') unless separator == ','
      scanner.skip(/\s*/)
    end
  when '['
    scanner.getch
    scanner.skip(/\s*/)
    return scanner.getch if scanner.peek(1) == ']'
    index = 0
    loop do
      scalar_spans(scanner, path + [index], targets, spans)
      scanner.skip(/\s*/)
      separator = scanner.getch
      break if separator == ']'
      abort('missing design array comma') unless separator == ','
      index += 1
      scanner.skip(/\s*/)
    end
  else
    start = scanner.pos
    token = scanner.scan(/"(?:[^"\\]|\\.)*"|true|false|null|-?(?:0|[1-9][0-9]*)(?:\.[0-9]+)?(?:[eE][+-]?[0-9]+)?/m)
    abort('invalid design scalar') unless token
    spans[path] = { 'offset' => start, 'length' => scanner.pos - start, 'original' => token } if targets.key?(path)
  end
end

def value_at(root, path)
  path.reduce(root) { |value, key| value.fetch(key) }
end

def assign_at(root, path, value)
  parent = path[0...-1].reduce(root) { |node, key| node.fetch(key) }
  parent[path.last] = value
end

def typed_edges(issue)
  (issue['dependencies'] || []).map do |edge|
    { 'id' => edge.fetch('id'), 'dependency_type' => edge.fetch('dependency_type') }
  end
end

def normalize_historical_readback(issue, updates)
  expected = full_copy(issue)
  dependencies = expected['dependencies'] || []
  dependencies.each do |dependency|
    id = dependency.fetch('id')
    next unless updates.key?(id)
    before_type = dependency.fetch('dependency_type')
    updates.fetch(id).each do |field, value|
      abort("nested child field absent: #{id}.#{field}") unless dependency.key?(field)
      dependency[field] = value
    end
    abort("nested dependency type changed: #{id}") unless dependency.fetch('dependency_type') == before_type
  end
  expected
end
