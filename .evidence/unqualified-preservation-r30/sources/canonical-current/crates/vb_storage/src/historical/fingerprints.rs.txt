use vb_core::WorkflowDigest;
use vb_core::action_pack::{ExecutorVersionDigest, ReceiptSchemaDigest};

fn fingerprint(sources: &[&[u8]]) -> WorkflowDigest {
    let mut hasher = blake3::Hasher::new();
    sources.iter().for_each(|source| {
        hasher.update(source);
    });
    WorkflowDigest::from_bytes(hasher.finalize().into())
}

pub(crate) fn executor_version() -> ExecutorVersionDigest {
    ExecutorVersionDigest::from_digest(fingerprint(&[
        include_bytes!("inputs/crates/vb_storage/src/executor/register.rs.source"),
        include_bytes!("inputs/crates/vb_storage/src/executor/receipt.rs.source"),
        include_bytes!("inputs/crates/vb_storage/src/executor/authority.rs.source"),
        include_bytes!("inputs/crates/vb_storage/src/executor/conformance.rs.source"),
        include_bytes!("inputs/crates/vb_storage/src/executor/catalog.rs.source"),
    ]))
}

pub(crate) fn receipt_schema() -> ReceiptSchemaDigest {
    ReceiptSchemaDigest::from_digest(fingerprint(&[include_bytes!(
        "inputs/crates/vb_storage/src/executor/receipt.rs.source"
    )]))
}
