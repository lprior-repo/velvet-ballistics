#![forbid(unsafe_code)]
//! Versioned complete checkpoint payload for one retry boundary.

use core::num::NonZeroU16;

use serde::{Deserialize, Serialize};
use vb_core::action::ActionContract;
use vb_core::frame::{CheckpointStepState, RunFrame, RunFrameCheckpointV1};
use vb_core::ids::{RunId, StepIdx, WorkflowDigest};
use vb_core::value_store::{ValueStore, ValueStoreCheckpointV1};
use vb_core::workflow::CompiledWorkflow;

use crate::admission::RunAdmission;
use crate::primitives::collect::{CollectStates, CollectStatesCheckpointV1};
use crate::shard::types::RunState;

#[path = "run_checkpoint/support.rs"]
mod support;

/// Complete v1 state needed to restore a run at a retry boundary.
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub struct RunCheckpointV1 {
    /// Checkpoint format version.
    pub version: u16,
    /// Run whose state is captured.
    pub run_id: RunId,
    /// Accepted artifact identity used to load the immutable workflow.
    pub artifact_digest: WorkflowDigest,
    /// Compiled workflow identity checked on restore.
    pub workflow_digest: WorkflowDigest,
    /// Durable attempt represented by this checkpoint.
    pub run_attempt: NonZeroU16,
    /// Failed step the retry command is authorized to execute again.
    pub retry_step: StepIdx,
    /// Journal sequence immediately before the checkpoint is appended.
    pub prior_seq: u64,
    /// Full execution frame, including slots, taint, PC, counters, and every step state.
    pub frame: RunFrameCheckpointV1,
    /// Full cold value-store arenas and limits.
    pub value_store: ValueStoreCheckpointV1,
    /// Per-action-step attempt counters.
    pub action_attempts: Box<[u16]>,
    /// Admission authority captured with the state.
    pub admission: Option<RunAdmission>,
    /// Wall-clock run start in milliseconds since the UNIX epoch, as recorded on the run state.
    pub run_started_millis: Option<u64>,
    /// Pagination state and page lineage captured with the state.
    pub collect_states: CollectStatesCheckpointV1,
    /// Cumulative journal bytes for the run through the RetryCheckpoint event carrying this checkpoint, including that event's own encoded size.
    pub journal_bytes_written: u64,
    /// Validated action contracts bound to the accepted artifact.
    pub action_contracts: vb_storage::StoredActionRegistry,
}

/// Errors proving a checkpoint cannot authorize restoration.
#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
pub enum RunCheckpointV1Error {
    /// The checkpoint version is not supported.
    #[error("unsupported runtime checkpoint version")]
    UnsupportedVersion,
    /// Frame run identity differs from the checkpoint identity.
    #[error("checkpoint frame belongs to a different run")]
    RunIdentityMismatch,
    /// Admission run identity differs from the checkpoint identity.
    #[error("checkpoint admission belongs to a different run")]
    AdmissionRunMismatch,
    /// Retry checkpoint lacks admission authority.
    #[error("retry checkpoint has no accepted admission authority")]
    MissingAdmissionAuthority,
    /// Admission artifact identity differs from the checkpoint identity.
    #[error("checkpoint admission has a different artifact digest")]
    ArtifactIdentityMismatch,
    /// Compiled workflow identity differs from checkpoint metadata.
    #[error("checkpoint workflow digest mismatch")]
    WorkflowIdentityMismatch,
    /// Frame dimensions do not match the accepted workflow.
    #[error("checkpoint frame dimensions differ from the workflow")]
    FrameDimensionsMismatch,
    /// Retry step is outside the workflow or not failed in the captured frame.
    #[error("checkpoint retry step is not the failed step")]
    RetryStepMismatch,
    /// Action attempt counters do not cover every frame step.
    #[error("checkpoint action attempt count differs from frame step count")]
    ActionAttemptCountMismatch,
    /// Checkpoint action contracts differ from the independently loaded artifact registry.
    #[error("checkpoint action contracts differ from accepted artifact")]
    ActionContractsMismatch,
    /// The per-step action attempt counter cannot be advanced.
    #[error("checkpoint action attempt counter overflow")]
    ActionAttemptOverflow,
    /// A nested state DTO failed validation.
    #[error("nested checkpoint validation failed")]
    InvalidNestedState,
    /// The checkpoint payload could not be decoded from durable bytes.
    #[error("checkpoint payload decode failed")]
    DecodeError,
    /// A retained declaration has no supported canonical projection.
    #[error("unsupported checkpoint action classification {action:?}")]
    UnsupportedHistoricalClassification { action: vb_core::ActionId },
    /// Observed provider effects are forbidden by exhaustive classification.
    #[error("forbidden checkpoint action classification {action:?}/{side_effect:?}")]
    ForbiddenAttestedClassification {
        action: vb_core::ActionId,
        side_effect: vb_core::SideEffect,
    },
    /// Retry restoration lacks the required canonical operation-key authority.
    #[error("checkpoint retry idempotency refused: {0}")]
    Idempotency(vb_core::IdempotencyViolation),
}

impl RunCheckpointV1 {
    /// Captures every mutable run-state component after validating retry authority.
    pub fn capture(
        run_id: RunId,
        artifact_digest: WorkflowDigest,
        run_attempt: NonZeroU16,
        retry_step: StepIdx,
        prior_seq: u64,
        state: &RunState,
    ) -> Result<Self, RunCheckpointV1Error> {
        support::validate_capture_run(state, run_id)?;
        support::validate_capture_admission(state, run_id, artifact_digest)?;
        let (frame, action_attempts) = support::failed_retry_frame(state, retry_step)?;
        capture_checkpoint(
            CheckpointCapture {
                run_id,
                artifact_digest,
                run_attempt,
                retry_step,
                prior_seq,
                state,
            },
            frame,
            action_attempts,
        )
    }

    /// Validates all identities and nested state against the accepted compiled workflow.
    pub fn validate_against(
        &self,
        workflow: &CompiledWorkflow,
        artifact_digest: WorkflowDigest,
        accepted_action_contracts: &vb_storage::StoredActionRegistry,
    ) -> Result<(), RunCheckpointV1Error> {
        support::validate_checkpoint_identity(self, workflow, artifact_digest)?;
        support::validate_checkpoint_retry_authority(self, accepted_action_contracts)?;
        support::validate_checkpoint_nested_state(self, workflow)
    }

    /// Restores a validated checkpoint using the independently loaded accepted workflow.
    pub fn restore(
        self,
        workflow: CompiledWorkflow,
        artifact: &vb_storage::AcceptedArtifact,
    ) -> Result<(RunState, NonZeroU16, StepIdx, u64), RunCheckpointV1Error> {
        self.validate_against(&workflow, artifact.digest, &artifact.action_contracts)?;
        let verified = crate::cold_contracts::verify(artifact)
            .map_err(|_| RunCheckpointV1Error::ArtifactIdentityMismatch)?;
        let canonical = verified
            .try_canonical()
            .map_err(support::projection_error)?;
        support::validate_retry_policy(&workflow, self.retry_step, &canonical)?;
        let run_attempt = self.run_attempt;
        let retry_step = self.retry_step;
        let prior_seq = self.prior_seq;
        let state = support::restore_checkpoint_state(self, workflow, canonical)?;
        Ok((state, run_attempt, retry_step, prior_seq))
    }
}

#[derive(Clone, Copy)]
struct CheckpointCapture<'a> {
    run_id: RunId,
    artifact_digest: WorkflowDigest,
    run_attempt: NonZeroU16,
    retry_step: StepIdx,
    prior_seq: u64,
    state: &'a RunState,
}

fn capture_checkpoint(
    capture: CheckpointCapture<'_>,
    frame: RunFrameCheckpointV1,
    action_attempts: Box<[u16]>,
) -> Result<RunCheckpointV1, RunCheckpointV1Error> {
    let contracts =
        vb_storage::StoredActionRegistry::try_from_canonical(&capture.state.action_contracts)
            .map_err(support::projection_error)?;
    let checkpoint = checkpoint_from_capture(capture, frame, action_attempts, contracts);
    checkpoint.validate_against(
        &capture.state.workflow,
        capture.artifact_digest,
        &checkpoint.action_contracts,
    )?;
    Ok(checkpoint)
}

fn checkpoint_from_capture(
    capture: CheckpointCapture<'_>,
    frame: RunFrameCheckpointV1,
    action_attempts: Box<[u16]>,
    action_contracts: vb_storage::StoredActionRegistry,
) -> RunCheckpointV1 {
    RunCheckpointV1 {
        version: 3,
        run_id: capture.run_id,
        artifact_digest: capture.artifact_digest,
        workflow_digest: capture.state.workflow.digest(),
        run_attempt: capture.run_attempt,
        retry_step: capture.retry_step,
        prior_seq: capture.prior_seq,
        frame,
        value_store: capture.state.store.checkpoint_v1(),
        action_attempts,
        admission: capture.state.admission.clone(),
        run_started_millis: capture.state.run_started_millis,
        collect_states: capture.state.collect_states.checkpoint_v1(),
        journal_bytes_written: capture.state.journal_bytes_written,
        action_contracts,
    }
}

fn validate_admission(
    admission: &RunAdmission,
    run_id: RunId,
    artifact_digest: WorkflowDigest,
) -> Result<(), RunCheckpointV1Error> {
    if admission.run_id() != run_id {
        return Err(RunCheckpointV1Error::AdmissionRunMismatch);
    }
    if admission.artifact_digest() != artifact_digest {
        return Err(RunCheckpointV1Error::ArtifactIdentityMismatch);
    }
    Ok(())
}

/// Decodes a durable checkpoint payload with version-aware validation.
///
/// Reads the VARINT-encoded version field first to determine the checkpoint
/// format version before decoding the full payload. Rejects historical version
/// 1 checkpoints that lack the policy_pin field and version 2 checkpoints
/// that lack the run-start and journal-byte counters, which cannot be safely decoded
/// as the current RunCheckpointV1 layout.
pub fn decode_checkpoint_payload(payload: &[u8]) -> Result<RunCheckpointV1, RunCheckpointV1Error> {
    if payload.is_empty() {
        return Err(RunCheckpointV1Error::UnsupportedVersion);
    }
    // Use postcard's streaming decoder to read the version field.
    // Postcard uses variable-length integer encoding, so we cannot
    // simply read the first two bytes as a u16.
    let mut decoder = postcard::Deserializer::from_bytes(payload);
    let version: u16 = serde::Deserialize::deserialize(&mut decoder)
        .map_err(|_| RunCheckpointV1Error::UnsupportedVersion)?;
    match version {
        1 | 2 => Err(RunCheckpointV1Error::UnsupportedVersion),
        3 => {
            let (decoded, rest) = postcard::take_from_bytes::<RunCheckpointV1>(payload)
                .map_err(|_| RunCheckpointV1Error::DecodeError)?;
            if !rest.is_empty() {
                return Err(RunCheckpointV1Error::DecodeError);
            }
            Ok(decoded)
        }
        _ => Err(RunCheckpointV1Error::UnsupportedVersion),
    }
}
