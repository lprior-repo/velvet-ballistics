use std::{num::NonZeroUsize, path::Path, sync::Arc};

use vb_core::{
    ActionId, RunId, SlotIdx, SlotValue, StepIdx, Taint, WorkflowDigest,
    action::{ActionContract, ActionName, ActionOutputReady, Idempotency, RetrySafety, SideEffect},
    capability::CapabilitySet,
    policy::RuntimePolicy,
    workflow::{CompiledNodeKind, CompiledWorkflow, ResourceContract, WorkflowParts, node},
};
use vb_runtime::{journal::StorageRuntimeJournal, runtime::Runtime, shard::ShardConfig};
use vb_storage::{FjallJournal, JournalEvent, RunSnapshot};

fn workflow() -> Result<CompiledWorkflow, String> {
    let action = node(
        StepIdx::ZERO,
        CompiledNodeKind::Do {
            action: ActionId::new(0),
            input: SlotIdx::ZERO,
        },
        Some(SlotIdx::new(1)),
        Some(StepIdx::new(1)),
    );
    let wait = node(
        StepIdx::new(1),
        CompiledNodeKind::WaitUntil {
            deadline_slot: SlotIdx::ZERO,
        },
        None,
        Some(StepIdx::new(2)),
    );
    let finish = node(
        StepIdx::new(2),
        CompiledNodeKind::Finish {
            result: SlotIdx::new(1),
        },
        None,
        None,
    );
    let mut parts = WorkflowParts {
        semantic_source_identity: None,
        source_metadata: None,
        name: Box::from("incident-runtime-report"),
        digest: WorkflowDigest::from_bytes([0; 32]),
        nodes: Box::from([action, wait, finish]),
        expressions: Box::from([]),
        accessors: Box::from([]),
        constants: Box::from([]),
        slot_count: 2,
        symbols_count: 0,
        entry: StepIdx::ZERO,
        step_names: Box::from([]),
        resource_contract: ResourceContract::DEFAULT,
        input_slots: Box::from([vb_core::CompiledInputSlot {
            slot: SlotIdx::ZERO,
            kind: vb_core::InputSlotKind::I64,
            constraint: vb_core::CompiledInputConstraint::required(),
        }]),
    };
    let encoded = postcard::to_allocvec(&parts).map_err(|error| error.to_string())?;
    parts.digest = WorkflowDigest::from_bytes(*blake3::hash(&encoded).as_bytes());
    CompiledWorkflow::try_from_parts(parts).map_err(|error| format!("workflow invalid: {error:?}"))
}

fn action_contract() -> Result<ActionContract, String> {
    Ok(ActionContract {
        id: ActionId::new(0),
        name: ActionName::new("incident-runtime-action").map_err(|error| error.to_string())?,
        input_slot_count: 1,
        output_slot_count: 1,
        max_input_bytes: 1024,
        max_output_bytes: 1024,
        timeout_ms: 1000,
        idempotency: Idempotency::DeterministicPure,
        side_effect: SideEffect::Pure,
        retry_safety: RetrySafety::Idempotent,
        required_capabilities: Box::from([]),
    })
}

pub(super) fn create_incident_events(
    path: &Path,
    run: RunId,
    complete: bool,
) -> Result<Vec<JournalEvent>, String> {
    let journal = Arc::new(FjallJournal::open(path, None).map_err(|error| error.to_string())?);
    let workflow = workflow()?;
    let workflow_digest = workflow.digest();
    let contracts = Box::from([action_contract()?]);
    vb_storage::submit_artifact_with_contracts(
        &journal,
        &workflow,
        RuntimePolicy::Strict,
        &contracts,
    )
    .map_err(|error| error.to_string())?;
    let shard_count = NonZeroUsize::new(1).ok_or_else(|| String::from("shard count is zero"))?;
    let shared = StorageRuntimeJournal::shared_journaled(journal.clone());
    let mut runtime = Runtime::new(shard_count, ShardConfig::default(), shared)
        .map_err(|error| error.to_string())?;
    runtime
        .submit_direct_with_inputs_grants_and_contracts(
            run,
            workflow,
            Box::from([(SlotIdx::ZERO, SlotValue::I64(4_102_444_800_000))]),
            CapabilitySet::empty(),
            contracts,
        )
        .map_err(|error| error.to_string())?;
    runtime.tick_all().map_err(|error| error.to_string())?;
    let ticket = runtime
        .lookup_pending_action_ticket(run, u64::from(StepIdx::ZERO.get()))
        .ok_or_else(|| String::from("runtime action ticket missing"))?;
    let secret = SlotValue::I64(987_654_321);
    let encoded = postcard::to_allocvec(&secret).map_err(|error| error.to_string())?;
    let encoded_len = u32::try_from(encoded.len()).map_err(|error| error.to_string())?;
    if complete {
        runtime
            .complete_action_with_output(
                ticket,
                ActionOutputReady {
                    output_slot: SlotIdx::new(1),
                    value: secret,
                    taint: Taint::Secret,
                    encoded_len,
                },
            )
            .map_err(|error| error.to_string())?;
    }
    if complete {
        runtime.tick_all().map_err(|error| error.to_string())?;
        runtime.tick_all().map_err(|error| error.to_string())?;
        runtime.tick_all().map_err(|error| error.to_string())?;
        let has_wait = journal
            .events_for_run_full(run)
            .map_err(|error| error.to_string())?
            .iter()
            .any(|event| matches!(event, JournalEvent::WaitScheduledEvent { .. }));
        if !has_wait {
            return Err(String::from(
                "completed action did not reach the durable wait boundary",
            ));
        }
    }
    runtime.cancel_run(run).map_err(|error| error.to_string())?;
    runtime.tick_all().map_err(|error| error.to_string())?;
    let events = journal
        .events_for_run_full(run)
        .map_err(|error| error.to_string())?;
    if complete {
        let snapshot_seq = events
            .iter()
            .find_map(|event| {
                matches!(event, JournalEvent::ActionCompletedEnvelope { .. }).then_some(event.seq())
            })
            .ok_or_else(|| String::from("runtime completion sequence missing"))?;
        journal
            .put_snapshot(&RunSnapshot {
                run,
                seq: snapshot_seq,
                workflow: workflow_digest,
                slots: Vec::new(),
                taint: Vec::new(),
            })
            .map_err(|error| error.to_string())?;
    }
    let expected_phase = if complete {
        events.iter().any(|event| {
            matches!(event,
            JournalEvent::ActionCompletedEnvelope { ticket: found, value, taint: Taint::Secret, .. }
                if *found == ticket && *value == encoded)
        })
    } else {
        events.iter().any(|event| {
            matches!(event,
            JournalEvent::ActionAbandoned { ticket: found, .. } if *found == ticket)
        })
    };
    if !expected_phase {
        return Err(String::from(
            "runtime did not persist the expected action phase",
        ));
    }
    drop(runtime);
    drop(journal);
    Ok(events)
}
