use super::*;
use vstd::prelude::*;

verus! {

pub(crate) open spec fn walk_seen(visited: Set<usize>, at: int) -> bool {
    exists|raw: usize| raw as int == at && visited.contains(raw)
}

pub(crate) proof fn walk_seen_insert(visited: Set<usize>, index: usize, at: int)
    ensures
        walk_seen(visited.insert(index), at) == (walk_seen(visited, at) || at == index as int),
{
    if walk_seen(visited.insert(index), at) {
        let raw = choose|raw: usize| raw as int == at && visited.insert(index).contains(raw);
        if raw != index {
            assert(visited.contains(raw));
            assert(walk_seen(visited, at));
        }
    }
    if walk_seen(visited, at) {
        let raw = choose|raw: usize| raw as int == at && visited.contains(raw);
        assert(visited.insert(index).contains(raw));
        assert(walk_seen(visited.insert(index), at));
    }
    if at == index as int {
        assert(visited.insert(index).contains(index));
        assert(walk_seen(visited.insert(index), at));
    }
}

pub(crate) open spec fn legacy_reference(
    p: &LegacyParts,
    keys: Set<u16>,
    s: Seq<usize>,
    stop: usize,
    outer: u32,
    v: Set<usize>,
    fuel: int,
    halt: bool,
) -> Option<u32>
    decreases fuel,
{
    if fuel <= 0 || s.len() == 0 {
        None
    } else {
        let idx = s.last();
        match projected_result(p, idx, stop, outer, keys, v) {
            Visit::Found(step) => Some(step as u32),
            Visit::Overflow if halt => Some(outer),
            _ => legacy_reference(
                p,
                keys,
                next_stack(p, idx, stop, outer, keys, v, s.drop_last()),
                stop,
                outer,
                next_visited(p, idx, stop, v),
                fuel - 1,
                halt,
            ),
        }
    }
}

pub(crate) open spec fn width(p: &LegacyParts, i: int) -> nat {
    decoded(
        p.nodes_meta@[i],
        if 0 <= i < p.kind_targets@.len() {
            p.kind_targets@[i]@
        } else {
            Seq::empty()
        },
        Seq::empty(),
    ).len()
}

pub(crate) open spec fn legacy_weight(p: &LegacyParts, v: Set<usize>, i: int) -> nat
    decreases p.nodes_meta@.len() - i,
{
    if i < 0 || i >= p.nodes_meta@.len() {
        0nat
    } else {
        (if walk_seen(v, i) {
            0nat
        } else {
            width(p, i)
        }) + legacy_weight(p, v, i + 1)
    }
}

pub(crate) proof fn insert_weight(p: &LegacyParts, v: Set<usize>, idx: usize, i: int)
    requires
        0 <= i <= p.nodes_meta@.len(),
    ensures
        legacy_weight(p, v, i) == legacy_weight(p, v.insert(idx), i) + if i <= idx
            < p.nodes_meta@.len() && !v.contains(idx) {
            width(p, idx as int)
        } else {
            0nat
        },
    decreases p.nodes_meta@.len() - i,
{
    if i < p.nodes_meta@.len() {
        insert_weight(p, v, idx, i + 1);
        walk_seen_insert(v, idx, i);
    }
}

pub(crate) proof fn projected_potential(
    p: &LegacyParts,
    k: Set<u16>,
    tail: Seq<usize>,
    idx: usize,
    stop: usize,
    outer: u32,
    v: Set<usize>,
)
    ensures
        legacy_weight(p, next_visited(p, idx, stop, v), 0) + next_stack(
            p,
            idx,
            stop,
            outer,
            k,
            v,
            tail,
        ).len() <= legacy_weight(p, v, 0) + tail.len(),
{
    insert_weight(p, v, idx, 0);
    if applicable(p, idx, stop, v) && idx <= 65535 && !projected_found(k, idx, outer) {
        let m = p.nodes_meta@[idx as int];
        let t = target_row(p, idx);
        assert(decoded(m, t, tail) =~= tail + decoded(m, t, Seq::empty()));
    }
}

pub(crate) fn legacy_walk(
    p: &LegacyParts,
    k: &std::collections::BTreeSet<u16>,
    start: usize,
    stop: usize,
    outer: u32,
    halt: bool,
) -> (out: Option<u32>)
    ensures
        out == legacy_reference(
            p,
            k@,
            seq![start],
            stop,
            outer,
            Set::empty(),
            1 + legacy_weight(p, Set::empty(), 0) as int,
            halt,
        ),
{
    let mut stack: Vec<usize> = Vec::new();
    stack.push(start);
    let mut v: std::collections::BTreeSet<usize> = std::collections::BTreeSet::new();
    let ghost mut budget = 1 + legacy_weight(p, Set::empty(), 0) as int;
    let ghost expected = legacy_reference(
        p,
        k@,
        seq![start],
        stop,
        outer,
        Set::empty(),
        budget,
        halt,
    );
    assert(stack@ =~= seq![start]);
    while !stack.is_empty()
        invariant
            budget >= 0,
            stack@.len() + legacy_weight(p, v@, 0) <= budget,
            expected == legacy_reference(
                p,
                k@,
                seq![start],
                stop,
                outer,
                Set::empty(),
                1 + legacy_weight(p, Set::empty(), 0) as int,
                halt,
            ),
            expected == legacy_reference(p, k@, stack@, stop, outer, v@, budget, halt),
        decreases budget,
    {
        let ghost prev = stack@;
        let ghost seen = v@;
        let Some(idx) = stack.pop() else {
            return None;
        };
        assert(idx == prev.last());
        assert(stack@ == prev.drop_last());
        assert(budget > 0);
        proof {
            projected_potential(p, k@, stack@, idx, stop, outer, v@);
        }
        let step = projection_visit(p, k, idx, stop, outer, &mut v, &mut stack);
        match step {
            Visit::Found(s) => return Some(u32::from(s)),
            Visit::Overflow if halt => return Some(outer),
            _ => {},
        }
        proof {
            reveal(legacy_reference);
            budget = budget - 1;
        }
    }
    None
}

} // verus!
